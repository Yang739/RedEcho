// app.js
const { ENV, DEBUG } = require('./config.js')
const store = require('./store.js')

App({
  globalData: {
    env: ENV,
    userInfo: null
  },

  // 全局错误捕获：把运行时异常 / 未处理的 Promise 拒绝暴露到 vConsole，
  // 便于定位"页面无响应 / 数据不出现"等只在真机/模拟器才暴露的问题。
  onError(err) {
    console.error('[RedEcho] 运行时错误:', err)
  },
  onUnhandledRejection(res) {
    console.error('[RedEcho] 未处理的 Promise 拒绝:', res && res.reason)
  },

  onLaunch() {
    if (!wx.cloud) {
      console.error('[RedEcho] 当前基础库版本过低，请使用 2.2.3 及以上的基础库以使用云能力')
      return
    }

    // 占位符环境 ID 表示使用本地 mock 调试，无需初始化云开发（避免控制台报 env 不存在）
    if (ENV === 'your-cloud-env-id') {
      if (DEBUG) console.log('[RedEcho] 未配置云环境，使用本地 mock 数据调试')
    } else {
      wx.cloud.init({
        // 使用你在「云开发控制台 - 设置」中看到的环境 ID
        env: ENV,
        // 在用户管理中记录访问用户，便于排查
        traceUser: true
      })
      if (DEBUG) {
        console.log('[RedEcho] 云开发环境已初始化，env =', ENV)
      }
    }

    // 启动分流：已登录工作人员直接进入自己主页（与游客启动进 pages/home/home 对应）
    if (store.getState().isStaff) {
      wx.reLaunch({ url: '/pages/staff/home/home' })
    }
  }
})

const TODAY_KEY = 'rede_today'
const POINT_LOG_KEY = 'rede_points_log'  // 积分明细（流水）记录
const POINTS_INIT_KEY = 'rede_points_init' // 迁移标记：旧版曾把种子 1280 写入本地，升级后统一归零
const SEED_POINTS = 0

function todayStr() {
  const d = new Date()
  const p = (n) => (n < 10 ? '0' + n : '' + n)
  return d.getFullYear() + '-' + p(d.getMonth() + 1) + '-' + p(d.getDate())
}

// 首次使用（或旧版升级）写入初始积分；旧版曾把种子 1280 写入本地存储，这里统一归零。
function loadPoints() {
  if (storage.get(POINTS_INIT_KEY, null) !== 'v1') {
    storage.set(POINTS_INIT_KEY, 'v1')
    storage.set(POINTS_KEY, SEED_POINTS)
    return SEED_POINTS
  }
  const v = storage.get(POINTS_KEY, null)
  if (typeof v === 'number') return v
  return SEED_POINTS

/* ---------- 第 8 页 ---------- */
}

function loadToday() {
  const v = storage.get(TODAY_KEY, null)
  if (v && typeof v === 'object' && v.date === todayStr()) {
    return typeof v.gain === 'number' ? v.gain : 0
  }
  return 0
}

// 经验（累计获得的积分）：仅正向奖励累加，消费不减少。
// 首次迁移时把当前积分存量作为经验起点（存量积分即历史累计获得），避免老用户等级从 0 起跳。
function loadExp(pointsSnapshot) {
  if (storage.get(EXP_INIT_KEY, null) !== 'v1') {
    storage.set(EXP_INIT_KEY, 'v1')
    const init = typeof pointsSnapshot === 'number' ? pointsSnapshot : 0
    storage.set(EXP_KEY, init)
    return init
  }
  const v = storage.get(EXP_KEY, null)
  return typeof v === 'number' ? v : 0
}

function loadLog() {
  const v = storage.get(POINT_LOG_KEY, [])
  return Array.isArray(v) ? v : []
}

const points0 = loadPoints()
const state = {
  user: null,        // { openid, nickname, avatar, level, isStaff }
  points: points0,            // 当前积分余额（持久化）

const subs = new Set()

function notify() {
  subs.forEach((fn) => {
    try { fn(state) } catch (e) { console.error('[store] listener error', e) }
  })
}

// 跨日则重置今日获得
function ensureToday() {
  const t = todayStr()
  if (state.todayDate !== t) {
    state.todayDate = t
    state.todayGain = 0
    storage.set(TODAY_KEY, { date: t, gain: 0 })
  }
}

module.exports = {
  getState() { return state },

  getLog() { return state.log },

  setUser(user) {
    Object.assign(state, user)
    state.loaded = true
    notify()
  },

  // delta 为积分变动量（正=获得，负=消费）；opts 为可选的明细描述：
  //   { type: 'read'|'show'|'postcard'|'redeem'|'adjust', title: '阅读文章', ref: id }

/* ---------- 第 10 页 ---------- */
  addPoints(delta, opts) {
    state.points = Math.max(0, state.points + delta)
    // 仅正向奖励累计「今日获得」与「经验」；消费（负向）不影响今日获得、也不扣减经验
    if (delta > 0) {
      ensureToday()
      state.todayGain += delta
      storage.set(TODAY_KEY, { date: state.todayDate, gain: state.todayGain })
      state.exp = (state.exp || 0) + delta
      storage.set(EXP_KEY, state.exp)
    }
    storage.set(POINTS_KEY, state.points)
    // 记录积分明细（流水）
    const o = opts || {}
    const entry = {
      id: 'pl' + Date.now() + Math.random().toString(36).slice(2, 6),
      type: o.type || 'adjust',
      title: o.title || (delta > 0 ? '积分奖励' : '积分消费'),
      ref: o.ref || '',
      delta: delta,
      balance: state.points,
      time: Date.now()
    }
    const log = state.log.slice()
    log.unshift(entry)
    if (log.length > LOG_MAX) log.length = LOG_MAX
    state.log = log
    storage.set(POINT_LOG_KEY, log)
    notify()
  },

  setPoints(points) {
    state.points = points
    storage.set(POINTS_KEY, state.points)
    notify()
  },

  // 更新个人资料（昵称 / 头像 / 签名 / 性别 / 生日 / 手机 / 密码）：合并后持久化
  setProfile(patch) {
    if (!patch || typeof patch !== 'object') return
    state.profile = Object.assign({}, state.profile, patch)
    storage.set(PROFILE_KEY, state.profile)
    notify()
  },

  // 切换为操作人员（场务/演员）：写入登录态并持久化
  setStaff(info) {
    if (!info || !info.phone) return
    state.staff = Object.assign({}, info)
    state.isStaff = true
    storage.set(STAFF_KEY, state.staff)

/* ---------- 第 11 页 ---------- */
    notify()
  },

  // 退出操作人员：清除登录态
  clearStaff() {
    state.staff = null
    state.isStaff = false
    storage.set(STAFF_KEY, null)
    notify()
  },

  subscribe(fn) {
    subs.add(fn)
    // 立即回放一次当前状态
    fn(state)
    return () => subs.delete(fn)
  },

    state.points = 0
    state.exp = 0
    state.isStaff = false

// 调用云函数的统一封装：成功取 result.data；result.code !== 0 视为业务错误
function call(name, data) {
  return new Promise((resolve, reject) => {
    wx.cloud.callFunction({
      name,
      data: data || {},
      success: (res) => {
        const r = res.result || {}
        if (r.code === 0) resolve(r.data)
        else reject(new Error(r.msg || ('cloud function ' + name + ' failed')))
      },
      fail: (err) => reject(err)
    })
  })
}

function delay(data, ms = 420) {
  return new Promise((resolve) => setTimeout(() => resolve(JSON.parse(JSON.stringify(data))), ms))
}

// 演出按开始时间排序：date 形如 "2026-07-18 19:30" 或 "2026-07-18"。
// 用 replace(' ','T') 兼容 iOS 端 Date 解析空格格式可能返回 NaN 的坑。
function showTimeOf(s) {
  if (!s || !s.date) return 0
  const t = new Date(String(s.date).replace(' ', 'T')).getTime()
  return isNaN(t) ? String(s.date) : t
}
function sortShows(list, filter) {
  const cmp = (a, b) => {
    const ta = showTimeOf(a), tb = showTimeOf(b)
    if (typeof ta === 'string' || typeof tb === 'string') {
      return String(ta) < String(tb) ? -1 : String(ta) > String(tb) ? 1 : 0
    }
    return ta - tb
  }
  const arr = (list || []).slice()
  if (filter !== 'all') return arr.sort(cmp)
  // 「全部」：非已结束按时间在前；已结束单独按时间排在末尾
  const active = arr.filter((s) => s.status !== 'ended').sort(cmp)
  const ended = arr.filter((s) => s.status === 'ended').sort(cmp)

/* ---------- 第 13 页 ---------- */
  return active.concat(ended)
}

const KEY = 'rede_favorites'
const MAX = 200

function getList() {
  try {
    const v = wx.getStorageSync(KEY)
    return Array.isArray(v) ? v.slice() : []
  } catch (e) {
    return []
  }
}

function has(id) {
  return getList().indexOf(id) !== -1
}

// 切换收藏状态：已收藏则取消，未收藏则加入最前；返回切换后的收藏状态（true=已收藏）
function toggle(id) {
  if (!id) return false
  let list = getList()
  const i = list.indexOf(id)
  if (i !== -1) list.splice(i, 1)
  else {
    list.unshift(id)
    if (list.length > MAX) list = list.slice(0, MAX)
  }
  try { wx.setStorageSync(KEY, list) } catch (e) {}

function calc(exp) {
  exp = Math.max(0, Math.floor(exp || 0))
  let cur = TIERS[0]
  let next = null
  for (let i = 0; i < TIERS.length; i++) {
    if (exp >= TIERS[i].exp) {
      cur = TIERS[i]
      next = TIERS[i + 1] || null
    } else {
      break
    }
  }
  const isMax = !next
  const curExp = cur.exp
  const nextExp = next ? next.exp : curExp
  const span = nextExp - curExp
  const inSpan = exp - curExp
  const pct = isMax ? 100 : Math.min(100, Math.round((inSpan / span) * 100))
  const gap = isMax ? 0 : (nextExp - exp)
  return {
    lv: cur.lv,
    name: cur.name,
    exp: exp,
    curExp: curExp,
    nextExp: next ? next.exp : null,
    nextName: next ? next.name : null,
    gap: gap,
    pct: pct,
    isMax: isMax
  }

// 将文本做「均衡分行」：行数尽量少，且相邻行字数差不超过 1
function splitBalanced(text, perLine) {
  const n = Math.max(1, Math.ceil(text.length / perLine))
  const base = Math.floor(text.length / n)
  const rem = text.length - base * n
  const lines = []
  let i = 0
  for (let k = 0; k < n; k++) {
    const take = base + (k < rem ? 1 : 0)
    lines.push(text.substr(i, take))
    i += take
  }
  return lines
}

// 以换行符断行（换行本身不显示）；所有句子（行/列）结尾都不跟标点；过长子段再均衡拆分
function splitQuoteLines(text, arrange, perLine) {
  let segs = String(text).split(/\r?\n/).map(s => s.trim()).filter(s => s.length)
  if (!segs.length) segs = [String(text).trim() || String(text)]
  // 句尾不跟标点：去除段尾全角/半角标点（横竖排一致）
  segs = segs.map(s => s.replace(/[。！？!?；：、，,.…—~·]+$/, ''))
  const lines = []
  for (const s of segs) {
    if (s.length <= perLine) lines.push(s)
    else lines.push(...splitBalanced(s, perLine))
  }
  return lines
}

// 日期统一为 YYYY.MM.DD（落库为 YYYY-MM-DD，预览/导出一致）

  if (m) return m[1] + '.' + m[2] + '.' + m[3]
  return d
}

async function saveImage(base64) {
  const buf = Buffer.from(base64, 'base64')
  const cloudPath = `postcards/${Date.now()}.png`
  const up = await cloud.uploadFile({ cloudPath, fileContent: buf })
  if (!up || !up.fileID) return ''
  const tmp = await cloud.getTempFileURL({ fileList: [up.fileID] })
  const item = tmp && tmp.fileList && tmp.fileList[0]
  return (item && item.tempFileURL) || ''
}
