// 职责：演员 AI 台词助手 —— 结合角色/情境生成多条备选台词。

// 真实接入（腾讯云混元大模型）：

const SECRET_ID = process.env.TENCENT_SECRET_ID
const SECRET_KEY = process.env.TENCENT_SECRET_KEY
const REGION = process.env.TENCENT_REGION || 'ap-guangzhou'
const MODEL = process.env.HUNYUAN_MODEL || 'hunyuan-standard'

// 从大模型文本中解析出台词数组（兼容 JSON / 退化分行）
function parseLines(content) {
  if (!content) return null
  let arr = extractJsonArray(content)
  if (Array.isArray(arr) && arr.length) {
    const lines = arr.slice(0, 5).map((it) => ({
      text: String(it.text || it.content || '').trim(),
      tone: String(it.tone || it.emotion || '深情').trim(),
      reason: String(it.reason || it.note || '').trim()
    })).filter((l) => l.text)
    if (lines.length) return lines
  }
  // 退化：按行拆分
  return content.split('\n').map((s) => s.trim()).filter(Boolean).slice(0, 5)
    .map((text) => ({ text, tone: '深情', reason: '' }))
}

function extractJsonArray(s) {
  try {
    const m = s.match(/\[[\s\S]*\]/)
    if (m) return JSON.parse(m[0])
  } catch (e) { /* ignore */ }
  try { return JSON.parse(s) } catch (e) { /* ignore */ }
  return null
}

// 调用腾讯云混元大模型，返回台词数组或 null（走回退）
async function callHunyuan(role, scene, situation) {
  if (!SECRET_ID || !SECRET_KEY) return null
  let hunyuan
  try {
    hunyuan = require('tencentcloud-sdk-nodejs-hunyuan')
  } catch (e) {
    console.warn('[aiLineGenerate] 未安装 tencentcloud-sdk-nodejs-hunyuan，回退示例台词')
    return null
  }
  const client = new hunyuan.hunyuan.v20230901.Client({
    credential: { secretId: SECRET_ID, secretKey: SECRET_KEY },
    region: REGION,
    profile: { httpProfile: { endpoint: 'hunyuan.tencentcloudapi.com' } }
  })
  const prompt =
    `你是沉浸式红色实景演出的编剧助手。请为以下设定生成 3 条可选台词，` +
    `兼顾戏剧张力与史实基调，并标注每条的情绪基调与创作意图。\n` +
    `角色：${role || '主角'}\n场景：${scene || '未知'}\n情境：${situation || ' unspecified'}\n` +
    `请严格以 JSON 数组返回，每条格式：{"text":"台词正文","tone":"情绪基调","reason":"创作意图"}，` +
    `不要输出任何额外说明文字。`
  const resp = await client.ChatCompletions({
    Model: MODEL,
    Messages: [{ Role: 'user', Content: prompt }],
    Stream: false
  })
  const content = resp && resp.Choices && resp.Choices[0] && resp.Choices[0].Message && resp.Choices[0].Message.Content
  return parseLines(content)
}

// 云函数：createOrder
// 职责：周边商城下单（积分/混合支付，生成核销码）。
// 写入集合：orders
const cloud = require('wx-server-sdk')
const crypto = require('crypto')
cloud.init({ env: cloud.DYNAMIC_CURRENT_ENV })
const db = cloud.database()
const _ = db.command
const getOpenid = () => cloud.getWXContext()
const ok = (data) => ({ code: 0, data })
const fail = (a, b = 1, extra) => { const code = (typeof a === 'number') ? a : b; const msg = (typeof a === 'number') ? b : a; return extra ? { code, msg, ...extra } : { code, msg } }

function genCode() {
  const ts = Date.now().toString(36).toUpperCase()
  const rnd = crypto.randomBytes(3).toString('hex').toUpperCase()
  return 'R' + ts + rnd
}

exports.main = async (event, context) => {
  const { OPENID } = getOpenid()
  if (!OPENID) return fail(401, 'missing openid')

  const { productId, title, spec, price, payType } = event
  if (!productId || price === undefined) return fail(400, 'missing productId or price')

  try {
    const now = new Date()
    const addRes = await db.collection('orders').add({
      data: {
        openid: OPENID,
        productId,
        title: title || '商品',
        spec: spec || '',
        price: price || 0,
        payType: payType || 'points',
        status: '待核销',
        redemptionCode: genCode(),
        created_at: now
      }
    })
    const order = (await db.collection('orders').doc(addRes._id).get()).data
    return ok(order)
  } catch (e) {
    return fail(500, 'createOrder failed: ' + e.message)
  }
}

// === cloudfunctions/getHomeFeed/index.js ===
// 云函数：getHomeFeed
// 职责：返回首页卡片列表 + 用户积分 + 今日获得积分 + 阅读进度 + 直播中的演出。
// 读取集合：cards / users / points_logs / shows（集合名全项目统一小写驼峰）。
const cloud = require('wx-server-sdk')
cloud.init({ env: cloud.DYNAMIC_CURRENT_ENV })
const db = cloud.database()
const _ = db.command
const getOpenid = () => cloud.getWXContext()
const ok = (data) => ({ code: 0, data })
const fail = (a, b = 1, extra) => { const code = (typeof a === 'number') ? a : b; const msg = (typeof a === 'number') ? b : a; return extra ? { code, msg, ...extra } : { code, msg } }

function greeting() {
  const h = new Date().getHours()
  if (h < 6) return '凌晨好'
  if (h < 12) return '上午好'
  if (h < 14) return '中午好'
  if (h < 18) return '下午好'
  return '晚上好'
}

exports.main = async (event, context) => {
  const { OPENID } = getOpenid()
  if (!OPENID) return fail(401, 'missing openid')

  try {
    // 用户积分
    const uRes = await db.collection('users').where({ openid: OPENID }).get()
    const points = uRes.data[0] ? (uRes.data[0].points || 0) : 0

    // 今日获得积分（points_logs 中 created_at 当天且 delta>0）
    const start = new Date()
    start.setHours(0, 0, 0, 0)
    const pRes = await db
      .collection('points_logs')
      .where({ openid: OPENID, delta: _.gt(0), created_at: _.gte(start) })
      .get()
    const todayGain = pRes.data.reduce((s, r) => s + (r.delta || 0), 0)

    // 推荐卡片（取前 3 篇，按 sort 升序）
    const cRes = await db
      .collection('cards')
      .orderBy('sort', 'asc')
      .limit(3)
      .get()

    // 阅读进度（reading 集合；未建则返回空）
    let reading = []
    try {
      const rRes = await db
        .collection('reading')
        .where({ openid: OPENID })
        .orderBy('progress', 'desc')
        .limit(5)
        .get()
      reading = rRes.data
    } catch (e) {
      reading = []
    }

    // 直播中的演出（取第一个 live）
    const sRes = await db
      .collection('shows')
      .where({ status: 'live' })
      .limit(1)
      .get()
    const liveShow = sRes.data[0] || null

    return ok({
      greeting: greeting(),
      points,
      todayGain,
      recommend: cRes.data,
      reading,
      liveShow
    })
  } catch (e) {
    return fail(500, 'getHomeFeed failed: ' + e.message)
  }

const ok = (data) => ({ code: 0, data })

exports.main = async (event, context) => {
  const { OPENID } = getOpenid()
  const { showId, optionId } = event
  if (!OPENID) return fail(401, 'missing openid')
  if (!showId || !optionId) return fail(400, 'missing showId or optionId')

  try {
    // 一人一票校验
    const voted = await db
      .collection('vote_records')
      .where({ openid: OPENID, showId })
      .count()
    if (voted.total > 0) {
      return fail(409, 'already voted', { voted: true })
    }

    // 找到该场次的投票题目
    const pRes = await db
      .collection('polls')
      .where({ showId, open: true })
      .limit(1)
      .get()
    if (!pRes.data[0]) return fail(404, 'no active poll')

    const pollId = pRes.data[0]._id
    const options = pRes.data[0].options || []
    const idx = options.findIndex((o) => o.id === optionId)
    if (idx < 0) return fail(400, 'invalid optionId')

    // 原子自增该选项票数
    await db
      .collection('polls')
      .doc(pollId)
      .update({
        data: { [`options.${idx}.votes`]: _.inc(1) }
      })

    // 落库投票记录
    await db.collection('vote_records').add({
      data: {
        openid: OPENID,
        showId,
        optionId,
        created_at: new Date()
      }
    })

    // 返回最新票数
    const fresh = (await db.collection('polls').doc(pollId).get()).data
    return ok({ voted: true, optionId, poll: fresh })
  } catch (e) {
    return fail(500, 'pollVote failed: ' + e.message)
  }
}

// 调用腾讯云 AI 绘画（文生图），返回 base64 或 null（走回退）
async function callTextToImage(prompt) {
  if (!SECRET_ID || !SECRET_KEY) return null
  let aiart
  try {
    aiart = require('tencentcloud-sdk-nodejs-aiart')
  } catch (e) {
    console.warn('[postcardGenerate] 未安装 tencentcloud-sdk-nodejs-aiart，回退 CSS 渲染')
    return null
  }
  const client = new aiart.aiart.v20221229.Client({
    credential: { secretId: SECRET_ID, secretKey: SECRET_KEY },
    region: REGION,
    profile: { httpProfile: { endpoint: 'aiart.tencentcloudapi.com' } }
  })
  const resp = await client.TextToImage({
    Prompt: prompt,
    RspImgType: 'base64'
  })
  return resp && resp.ResultImage ? resp.ResultImage : null
}

// 将 base64 上传云存储并返回可直接 <image> 显示的临时 URL
async function saveImage(base64) {

  if (!up || !up.fileID) return ''
  const tmp = await cloud.getTempFileURL({ fileList: [up.fileID] })
  const item = tmp && tmp.fileList && tmp.fileList[0]
  return (item && item.tempFileURL) || ''
}

// 读取集合：shows / vote_records / emotion_feedbacks / quotes
const cloud = require('wx-server-sdk')
cloud.init({ env: cloud.DYNAMIC_CURRENT_ENV })
const db = cloud.database()
const _ = db.command
const $ = db.command.aggregate
const getOpenid = () => cloud.getWXContext()
const ok = (data) => ({ code: 0, data })
const fail = (a, b = 1, extra) => { const code = (typeof a === 'number') ? a : b; const msg = (typeof a === 'number') ? b : a; return extra ? { code, msg, ...extra } : { code, msg } }

// 情绪类型（与 getShowRoom / 前端 re-tag 配色一致）
const EMOTIONS = [
  { id: 'e1', label: '感动' },
  { id: 'e2', label: '震撼' },
  { id: 'e3', label: '激昂' },
  { id: 'e4', label: '沉思' }
]
// 无数据时的默认分布（避免前端空白）
const DEFAULT_EMO = [
  { label: '感动', pct: 42 },
  { label: '震撼', pct: 31 },
  { label: '激昂', pct: 19 },
  { label: '沉思', pct: 8 }
]
const DEFAULT_QUOTE = '为有牺牲多壮志，敢教日月换新天。'

exports.main = async (event, context) => {
  const { OPENID } = getOpenid()
  if (!OPENID) return fail('missing openid', 401)

  const showId = event.showId
  if (!showId) return fail('missing showId', 400)

  try {
    // 场次
    const sRes = await db.collection('shows').where({ id: showId }).get()
    const show = sRes.data[0] || null

    // 我的投票（该用户在该场次的最新一票）
    let myVote = ''
    const vRes = await db
      .collection('vote_records')
      .where({ openid: OPENID, showId })
      .orderBy('created_at', 'desc')
      .limit(1)
      .get()
    if (vRes.data[0]) {
      const v = vRes.data[0]
      myVote = v.optionLabel || v.optionId || ''
    }

    // 情绪分布（按 type 聚合 → 百分比）
    let emotionDist = DEFAULT_EMO
    const agg = await db
      .collection('emotion_feedbacks')
      .aggregate()
      .match({ showId })
      .group({ _id: '$type', count: $.sum(1) })
      .end()
    if (agg.list && agg.list.length) {
      const total = agg.list.reduce((s, x) => s + x.count, 0)
      emotionDist = EMOTIONS.map((e) => {
        const hit = agg.list.find((x) => x._id === e.id)
        const c = hit ? hit.count : 0
        return { label: e.label, pct: total ? Math.round((c / total) * 100) : 0 }
      })
    }

    // 一句金句（随机取一条，兼容字符串/对象存储）
    let quote = DEFAULT_QUOTE
    const qAgg = await db.collection('quotes').aggregate().sample({ size: 1 }).end()
    if (qAgg.list && qAgg.list[0]) {
      const q = qAgg.list[0]
      quote = typeof q === 'string' ? q : (q.text || DEFAULT_QUOTE)
    }

    return ok({ show, myVote, emotionDist, quote })
  } catch (e) {
    return fail('showReview failed: ' + e.message, 500)
  }
}

// 复用 services/pcLayout.js 的 paintPostcard，将明信片以 Canvas 绘制。
// 展示统一为竖版(2:3)：横版内容在 canvas 内部顺时针旋转 90°（与导出一致）。
// <canvas> 元素始终 100% 填满父容器，无裁剪问题。
const { paintPostcard, decorateCard } = require('../../services/pcLayout.js')

// 参考分辨率：统一竖版 600×900（与 .pc-frame aspect-ratio: 2/3 对应）
const REF_W = 600
const REF_H = 900

function getDpr() {
  try {
    if (wx.getWindowInfo) return wx.getWindowInfo().pixelRatio || 2
    return wx.getSystemInfoSync().pixelRatio || 2
  } catch (e) { return 2 }
}

Component({
  properties: {
    card: { type: Object, value: null, observer() { this._render() } },
    // 覆盖长边上限 px（未乘 dpr）。预览传 900 高清；默认缩略图 480 控内存。
    res: { type: Number, value: 0 }
  },

  lifetimes: {
    attached() { this._canvas = null },
    ready() { this._query() }
  },

  pageLifetimes: {
    show() { this._render() } // 字体加载后重绘用上真·书法体
  },

  methods: {
    _query() {
      wx.createSelectorQuery().in(this).select('.cv')
        .fields({ node: true, size: true })
        .exec((res) => {
          if (res && res[0] && res[0].node) {
            this._canvas = res[0].node
            this._render()
          }
        })
    },

    _render() {
      const canvas = this._canvas
      const card = this.data.card
      if (!canvas || !card) return

      const orient = card.orient || 'land'
      // 长边上限：按竖版 900 为基准
      const maxLong = (this.data.res && this.data.res > 0) ? this.data.res : 480
      const scale = Math.min(maxLong / REF_W, maxLong / REF_H)

      const dpr = getDpr()
      const W = Math.round(REF_W * scale)
      const H = Math.round(REF_H * scale)
      canvas.width = W * dpr
      canvas.height = H * dpr

      const ctx = canvas.getContext('2d')
      ctx.setTransform(1, 0, 0, 1, 0, 0)
      ctx.clearRect(0, 0, canvas.width, canvas.height)
      ctx.scale(dpr * scale, dpr * scale)

      const c = decorateCard(card)
      if (orient === 'land') {
        // 横版：在 600×900 竖版画布内，translate(600,0)+rotate(π/2) 把 3:2 组合旋转填满
        ctx.translate(REF_W, 0)
        ctx.rotate(Math.PI / 2)
        paintPostcard(ctx, 900, 600, c)
      } else {
        paintPostcard(ctx, REF_W, REF_H, c)
      }
    }
  }
})

function genCode() {
  const ts = Date.now().toString(36).toUpperCase()
  const rnd = Math.random().toString(36).slice(2, 6).toUpperCase()
  return 'R' + ts + rnd
}

function getProducts() {
  return PRODUCTS
}

function getOrders() {
  const v = storage.get(ORDERS_KEY, [])
  return Array.isArray(v) ? v : []
}

// 兑换：校验库存(此处无限)/积分；成功则扣积分并写入订单
// 返回 { ok, order } 或 { ok:false, reason: 'notfound' | 'points' }
function purchase(productId, qty) {
  qty = qty || 1
  const p = PRODUCTS.find((x) => x.id === productId)
  if (!p) return { ok: false, reason: 'notfound' }
  const total = p.price * qty
  if (store.getState().points < total) return { ok: false, reason: 'points' }
  const order = {
    id: 'mo' + Date.now(),
    productId,
    title: p.title,
    desc: p.desc,
    price: p.price,
    qty,
    total,
    status: '待核销',
    code: genCode(),
    created_at: Date.now(),
    gradient: p.gradient
  }
  store.addPoints(-total, { type: 'redeem', title: '兑换 ' + p.title, ref: order.id })
  const list = getOrders()
  list.unshift(order)
  storage.set(ORDERS_KEY, list)
  return { ok: true, order }
}

const DEFAULT_QUOTE = ""
const SIGN_BAND = 96   // 底部落款带预留高度（Canvas px）
const SCENE_BAND = 112 // 顶部场景名预留高度（Canvas px）

// 将文本做「均衡分行」：行数尽量少，且相邻行字数差不超过 1
function splitBalanced(text, perLine) {
  const n = Math.max(1, Math.ceil(text.length / perLine))
  const base = Math.floor(text.length / n)
  const rem = text.length - base * n
  const lines = []
  let i = 0
  for (let k = 0; k < n; k++) {
    const take = base + (k < rem ? 1 : 0)
    lines.push(text.substr(i, take))
    i += take
  }
  return lines
}

// 以换行符断行（换行本身不显示）；所有句子（行/列）结尾都不跟标点；过长子段再均衡拆分
function splitQuoteLines(text, arrange, perLine) {
  let segs = String(text).split(/\r?\n/).map(s => s.trim()).filter(s => s.length)
  if (!segs.length) segs = [String(text).trim() || String(text)]
  // 句尾不跟标点：去除段尾全角/半角标点（横竖排一致）
  segs = segs.map(s => s.replace(/[。！？!?；：、，,.…—~·]+$/, ''))
  const lines = []
  for (const s of segs) {
    if (s.length <= perLine) lines.push(s)
    else lines.push(...splitBalanced(s, perLine))
  }
  return lines
}

// 日期统一为 YYYY.MM.DD（落库为 YYYY-MM-DD，预览/导出一致）
function formatDate(d) {
  if (!d) return ''
  const m = String(d).match(/^(\d{4})-(\d{2})-(\d{2})$/)
  if (m) return m[1] + '.' + m[2] + '.' + m[3]
  return d
}

// 计算名言的展示方案（与 Canvas 规则对齐：换行断行 / 句尾去标点 / 续行缩进）
function buildQuote(quote, orient, arrange, position) {
  const raw = (quote && String(quote).trim()) || DEFAULT_QUOTE
  // 以换行符断行、句尾去标点；空行忽略
  let lines = String(raw).split(/\r?\n/).map(s => s.trim()).filter(s => s.length)
    .map(s => s.replace(/[。！？!?；：、，,.…—~·]+$/, ''))
  if (!lines.length) lines = [raw]
  const full = !!POS_FULL[position]
  const indent = (position === 'left' || position === 'tl' || position === 'bl')
  const longest = lines.reduce((m, s) => Math.max(m, s.length), 0)

  if (arrange === 'v') {
    // 竖排：以最长段定字号（尽量一列放下）
    const maxCol = orient === 'port' ? 16 : 11
    let fs = 34
    if (longest > maxCol) fs = Math.max(22, Math.round((34 * maxCol) / longest))
    return { lines, fontSizeRpx: fs, indent }
  }

  // 横排：按位置估算单行容量（基准字号 34rpx，CJK 约等宽）
  const perLine = orient === 'port'
    ? (full ? 10 : 5)
    : (full ? 15 : 7)
  if (longest <= perLine) {
    const boost = longest <= Math.ceil(perLine * 0.6) ? 6 : 0
    return { lines, fontSizeRpx: Math.min(44, 34 + boost), indent }
  }
  const fs = Math.max(22, Math.round((34 * perLine) / longest))
  return { lines, fontSizeRpx: fs, indent }
}

// 补全一张明信片渲染所需字段
function decorateCard(rec) {
  const orient = rec.orient || 'land'
  const arrange = rec.arrange || 'h'
  const position = rec.position || 'center'
  const q = buildQuote(rec.quote, orient, arrange, position)
  return Object.assign({}, rec, {
    orient, arrange, position,
    quoteLines: q.lines,
    quoteFontSize: q.fontSizeRpx,
    indent: q.indent,
    sign: rec.sign || '红韵智演',
    date: formatDate(rec.date)
  })
}

// ---------------- Canvas 精绘 ----------------

function paintPostcard(ctx, W, H, rec) {
  const colors = (rec.colors && rec.colors.length === 2) ? rec.colors : ['#C8161D', '#7A0A10']
  const F = FONTS[rec.font] || FONTS.maoti
  const orient = rec.orient || 'land'
  const arrange = rec.arrange || 'h'
  const position = rec.position || 'center'

  // 背景
  const g = ctx.createLinearGradient(0, 0, W, H)
  g.addColorStop(0, colors[0])
  g.addColorStop(1, colors[1])
  ctx.fillStyle = g
  ctx.fillRect(0, 0, W, H)

  ctx.fillStyle = '#ffffff'

  // 场景名（左上）
  ctx.textBaseline = 'top'
  ctx.textAlign = 'left'
  ctx.font = F.weight + ' 40px ' + F.family
  ctx.fillText(rec.scene || '', 40, 40)

  // 落款（右下，双行：日期 / 红韵智演）
  ctx.textAlign = 'right'
  ctx.fillStyle = 'rgba(255,255,255,.9)'
  ctx.font = '400 22px sans-serif'
  ctx.fillText(rec.date || '', W - 40, H - 76)
  ctx.fillStyle = '#ffffff'
  ctx.font = '700 28px sans-serif'
  ctx.fillText(rec.sign || '红韵智演', W - 40, H - 44)

  // 名言（不加引号，按用户要求纯文本呈现）
  const quote = (rec.quote && String(rec.quote).trim()) ? rec.quote : DEFAULT_QUOTE
  const q = quote
  if (arrange === 'v') paintVertical(ctx, q, W, H, F, orient, position)
  else paintHorizontal(ctx, q, W, H, F, orient, position)
}

function paintHorizontal(ctx, q, W, H, F, orient, position) {
  const pad = 40
  // br（右下）多行时降级为 bottom 居中，避让右下落款带
  const eff = (position === 'br' && q.length > (orient === 'port' ? 9 : 14)) ? 'bottom' : position

  let boxX = pad
  let boxW = W - pad * 2
  let align = 'center'
  if (eff === 'left' || eff === 'tl' || eff === 'bl') { boxW = (W - pad * 2) * 0.62; align = 'left' }
  if (eff === 'right' || eff === 'tr') { boxX = pad + (W - pad * 2) * 0.38; boxW = (W - pad * 2) * 0.62; align = 'right' }

  // 字号自适应：先按换行切段、去句尾标点，以最长段为基准测量，超宽则缩（min 20）
  let fs = orient === 'port' ? 40 : 38
  ctx.font = F.weight + ' ' + fs + 'px ' + F.family
  const segs = splitQuoteLines(q, 'h', q.length)
  let longestW = 0
  for (const s of segs) longestW = Math.max(longestW, ctx.measureText(s).width)
  if (longestW > boxW) {
    fs = Math.max(20, Math.floor((fs * boxW) / longestW))
    ctx.font = F.weight + ' ' + fs + 'px ' + F.family
  }

  // 分行：以换行符断行；续行缩进两字；超长段再均衡折行
  const perLine = Math.max(1, Math.floor(boxW / (fs * 0.95)))
  let lines = splitQuoteLines(q, 'h', perLine)

  // 限行（最多 6 行）：超出再缩字号
  if (lines.length > 6) {
    fs = Math.max(18, Math.floor((fs * 6) / lines.length))
    ctx.font = F.weight + ' ' + fs + 'px ' + F.family
    const pl = Math.max(1, Math.floor(boxW / (fs * 0.95)))
    lines = splitQuoteLines(q, 'h', pl)
    if (lines.length > 6) lines = lines.slice(0, 6)
  }

  const lineH = fs * 1.5
  const totalH = lines.length * lineH
  const indentX = fs * 2 // 续行缩进两字

  // 垂直锚点（bottom 系预留落款带）
  let y
  if (eff === 'top' || eff === 'tl' || eff === 'tr') y = SCENE_BAND
  else if (eff === 'bottom' || eff === 'bl' || eff === 'br') y = H - SIGN_BAND - totalH
  else y = (H - totalH) / 2

  ctx.textBaseline = 'top'
  ctx.textAlign = align
  ctx.fillStyle = '#ffffff'
  for (let i = 0; i < lines.length; i++) {
    const baseX = align === 'center' ? boxX + boxW / 2
      : align === 'right' ? boxX + boxW
      : boxX
    // 续行相对上一行缩进两字（右对齐则向左缩）
    const x = baseX + (i > 0 ? (align === 'right' ? -indentX : indentX) : 0)
    ctx.fillText(lines[i], x, y + i * lineH)
  }
}

function paintVertical(ctx, q, W, H, F, orient, position) {
  const pad = 40
  // 横版竖向容量放宽，使单列可容纳约 14 汉字 + 1 逗号（约 15 字）；竖版保持更大留白
  const maxH = orient === 'port' ? H - 200 : H - 240
  // 先按换行切段、去段尾标点，以最长段为基准定字号（保证单列不超列高）
  const segs = splitQuoteLines(q, 'v', q.length)
  const maxSeg = segs.reduce((m, s) => Math.max(m, s.length), 0)
  let fs = 46
  const colCap = Math.floor(maxH / (fs * 1.05))
  if (maxSeg > colCap) fs = Math.max(22, Math.floor((fs * colCap) / maxSeg))
  ctx.font = F.weight + ' ' + fs + 'px ' + F.family
  const charH = fs * 1.05
  const cap = Math.floor(maxH / charH)

  // 每段再按列高切列（列序：续列朝中心方向展开）
  const cols = []
  for (const seg of segs) {
    let i = 0
    while (i < seg.length) { cols.push(seg.substr(i, cap)); i += cap }
  }

  // 水平锚点
  let x, colDir
  if (position === 'left' || position === 'tl' || position === 'bl') {
    x = pad + fs / 2
    colDir = 1
  } else if (position === 'center') {
    // 居中：以首列（最右列）中心为基准，整块列向中心展开后整体水平居中
    x = W / 2 + (cols.length - 1) * (fs * 1.15) / 2
    colDir = -1
  } else {
    x = W - pad - fs / 2
    colDir = -1
  }

  // 续列（换行后 / 段内换行）整体向下缩进两字
  const indentY = fs * 2
  const mainH = (cols[0] ? cols[0].length * charH : 0) + indentY

  // 垂直锚点（bottom 系预留落款带；以首列高度 + 续列缩进估算整体块高）
  let y0
  if (position === 'top' || position === 'tl' || position === 'tr') y0 = SCENE_BAND
  else if (position === 'bottom' || position === 'bl' || position === 'br') y0 = H - SIGN_BAND - mainH
  else y0 = (H - mainH) / 2

  // 竖版竖排（左锚点）单列下移，避让左上角场景名，不与其相接
  if (orient === 'port' && (position === 'left' || position === 'tl' || position === 'bl')) {
    y0 = Math.max(y0, SCENE_BAND + 16)
  }

  ctx.textBaseline = 'top'
  ctx.textAlign = 'center'
  ctx.fillStyle = '#ffffff'
  for (let c = 0; c < cols.length; c++) {
    const cx = x + colDir * c * (fs * 1.15)
    const col = cols[c]
    const yStart = y0 + (c > 0 ? indentY : 0)
    for (let j = 0; j < col.length; j++) {
      ctx.fillText(col[j], cx, yStart + j * charH)
    }
  }
}

// 把明信片绘制到离屏 canvas 并导出：统一为竖版(2:3)，
// 横版内容顺时针旋转 90° CW（展示态与导出均统一竖版 2:3、横版内旋），所见即所得。
// canvas：type=2d 的 canvas node；card：与 paintPostcard 同结构。
function renderPostcard(canvas, card) {
  const orient = (card && card.orient) || 'land'
  const dpr = (wx.getWindowInfo ? wx.getWindowInfo().pixelRatio : wx.getSystemInfoSync().pixelRatio) || 2
  const W = 600
  const H = 900
  canvas.width = W * dpr
  canvas.height = H * dpr
  const ctx = canvas.getContext('2d')
  // 重设 canvas.width 已重置变换矩阵；这里再显式归零，复用同一节点时安全
  ctx.setTransform(1, 0, 0, 1, 0, 0)
  ctx.clearRect(0, 0, canvas.width, canvas.height)
  ctx.scale(dpr, dpr)
  const c = decorateCard(card)
  if (orient === 'land') {
    // 横版 3:2 组合旋转 90° CW 放入 600×900 竖版画布：
    // 逻辑 900×600 → translate(600,0) + rotate(π/2) 恰好映射到 600×900
    ctx.translate(W, 0)
    ctx.rotate(Math.PI / 2)
    paintPostcard(ctx, 900, 600, c)
  } else {
    paintPostcard(ctx, W, H, c)
  }
  return canvas
}

module.exports = { buildQuote, decorateCard, paintPostcard, renderPostcard, DEFAULT_QUOTE }


// === services/postcards.js ===
// services/postcards.js —— 本地「我的作品」明信片存储
// 与 readHistory / favorites 一致：用 wx 本地存储持久化用户创作的明信片，
// 保证「已存入我的作品」提示真实可信，且创作页/我的页/作品列表页同源读取。

const KEY = 'rede_postcards'

function getList() {
  try {
    const v = wx.getStorageSync(KEY)
    return Array.isArray(v) ? v : []
  } catch (e) {
    return []
  }
}

function today() {
  const d = new Date()
  const p = (n) => (n < 10 ? '0' + n : '' + n)
  return d.getFullYear() + '-' + p(d.getMonth() + 1) + '-' + p(d.getDate())
}

// 新增一条作品（置顶），自动补 id / date
function add(rec) {
  const list = getList()
  const item = Object.assign({ id: 'p' + Date.now(), date: today() }, rec)
  list.unshift(item)
  try { wx.setStorageSync(KEY, list) } catch (e) {}
  return list
}

function remove(id) {
  const list = getList().filter((p) => p.id !== id)
  try { wx.setStorageSync(KEY, list) } catch (e) {}
  return list
}

function clear() {
  try { wx.setStorageSync(KEY, []) } catch (e) {}
  return []
}

// 存储键是否曾被写入：用于区分"从未初始化"与"用户已主动删空"
//（WeChat 中未设置的键 getStorageSync 返回 ''，已设为空数组则返回 []）
function isInitialized() {
  try {
    return wx.getStorageSync(KEY) !== ''
  } catch (e) {
    return false
  }
}

module.exports = { getList, add, remove, clear, today, isInitialized }


// === services/readHistory.js ===
// services/readHistory.js —— 本地阅读历史（"我的阅读"模块）
//
// 使用 wx 本地存储记录已读资料卡（含阅读时间），倒序、去重、限量，无需后端依赖。
// 游客端未接入账号体系前，阅读记录仅保存在本机，卸载小程序或清缓存会丢失。

const KEY = 'rede_read_history'
const MAX = 50

function getRaw() {
  try {
    const v = wx.getStorageSync(KEY)
    return Array.isArray(v) ? v : []
  } catch (e) {
    return []
  }
}

// 兼容旧数据（旧版仅存 id 字符串）：统一为 { id, time }
function normalize(list) {
  return list.map((e) => {
    if (typeof e === 'string') return { id: e, time: Date.now() }
    return { id: e.id, time: e.time || Date.now() }
  })
}

// 已读 id 列表（最新在前），供去重 / 已读状态判断使用
function getList() {
  return normalize(getRaw()).map((e) => e.id)
}

// 已读条目（含阅读时间），供列表展示使用
function getEntries() {
  return normalize(getRaw())
}

// 记录一次阅读：把 id 提到最前、刷新阅读时间、去重、限量
function add(id) {
  if (!id) return getEntries()
  const list = getRaw().filter((e) => (typeof e === 'string' ? e : e.id) !== id)
  list.unshift({ id, time: Date.now() })
  if (list.length > MAX) list = list.slice(0, MAX)
  try { wx.setStorageSync(KEY, list) } catch (e) {}
  return getEntries()
}

function clear() {
  try { wx.setStorageSync(KEY, []) } catch (e) {}
  return []
}

// 阅读时间格式化：2026 年 7 月 19 日  14 : 30（24 小时制）
function pad(n) {
  return n < 10 ? '0' + n : '' + n
}

function formatDate(time) {
  const d = new Date(time)
  const y = d.getFullYear()
  const m = d.getMonth() + 1
  const day = pad(d.getDate())
  const hh = pad(d.getHours())
  const mm = pad(d.getMinutes())
  return y + ' 年 ' + m + ' 月 ' + day + ' 日  ' + hh + ' : ' + mm
}

// 删除单条阅读记录（按 id），返回删除后的条目列表
function remove(id) {
  const list = getRaw().filter((e) => (typeof e === 'string' ? e : e.id) !== id)
  try { wx.setStorageSync(KEY, list) } catch (e) {}
  return getEntries()
}

module.exports = { getList, getEntries, add, remove, clear, formatDate }

// === services/rewards.js ===
// services/rewards.js —— 积分奖励与明信片计费
//
// 规则（与产品需求一致）：
//   1) 阅读一篇文章   +1 积分（同一篇只加一次，按已奖励集合去重）
//   2) 参加一次演出   +5 积分（同一场只加一次）
//   3) 明信片前 5 次生成免费；之后每次生成消耗 1 积分
//      - 免费额度用「累计生成次数」判定，删除作品不会回退已用掉的免费次数
//      - 计费用独立的生成计数器，避免「删了再生成」刷免费额度

const store = require('../store.js')
const storage = require('./storage.js')

const READ_KEY = 'rede_awarded_reads'   // 已奖励积分的文章 id 集合
const SHOW_KEY = 'rede_awarded_shows'   // 已奖励积分的演出 id 集合
const GEN_KEY = 'rede_postcard_count'   // 累计明信片生成次数
const POSTCARD_FREE = 5                 // 前 N 次免费

function getArr(key) {
  const v = storage.get(key, [])
  return Array.isArray(v) ? v : []
}
function pushArr(key, id) {
  const a = getArr(key)
  if (!a.includes(id)) {
    a.push(id)
    storage.set(key, a)
  }
}

// 阅读一篇文章 +1（同篇一次）
function awardRead(id) {
  if (!id) return false
  if (getArr(READ_KEY).includes(id)) return false
  pushArr(READ_KEY, id)
  store.addPoints(1, { type: 'read', title: '阅读文章', ref: id })
  return true
}

// 参加一次演出 +5（同场一次）
function awardShow(showId) {
  if (!showId) return false
  if (getArr(SHOW_KEY).includes(showId)) return false
  pushArr(SHOW_KEY, showId)
  store.addPoints(5, { type: 'show', title: '参与演出', ref: showId })
  return true
}

// 当前明信片计费状态（生成前查询，用于 UI 提示与拦截）
// 返回 { count, free, freeLeft }
function postcardState() {
  const n = getGenCount()
  return {
    count: n,
    free: n < POSTCARD_FREE,
    freeLeft: Math.max(0, POSTCARD_FREE - n)
  }
}

// 生成成功后提交：计数 +1；若已超过免费次数则扣 1 积分
// 返回 { free } —— 本次是否免费
function commitPostcard() {
  const n = getGenCount()
  const free = n < POSTCARD_FREE
  storage.set(GEN_KEY, n + 1)
  if (!free) store.addPoints(-1, { type: 'postcard', title: '生成明信片', ref: 'pc' + Date.now() })
  return { free }
}

function getGenCount() {
  const v = storage.get(GEN_KEY, 0)
  return typeof v === 'number' ? v : 0
}

module.exports = { awardRead, awardShow, postcardState, commitPostcard, POSTCARD_FREE }


// === services/storage.js ===
// services/storage.js —— 安全的 wx 本地存储封装
// 统一处理「键未设置」「非小程序环境（node 桩测）」等边界，避免各处重复写 try/catch。

function get(key, fallback) {
  try {
    if (typeof wx !== 'undefined' && wx.getStorageSync) {
      const v = wx.getStorageSync(key)
      if (v === '' || v === undefined || v === null) return fallback
      return v
    }
  } catch (e) {}
  return fallback
}

function set(key, val) {
  try {
    if (typeof wx !== 'undefined' && wx.setStorageSync) {
      wx.setStorageSync(key, val)
    }
  } catch (e) {}
}

module.exports = { get, set }


// === store.js ===
// store.js —— 轻量全局状态（用户 / 积分 / 今日获得 / 身份）
// 页面通过 subscribe 订阅，onUnload 时取消订阅，避免内存泄漏。
// 真实环境由云函数 login 写入 userInfo，这里先用本地兜底。
//
// 积分持久化：points 与 todayGain（今日获得）写入本地存储，跨页面 / 重启保留；
// todayGain 按自然日自动清零，仅统计当日通过奖励获得的积分（消费不计入）。

const storage = require('./services/storage.js')

const POINTS_KEY = 'rede_points'
const TODAY_KEY = 'rede_today'
const POINT_LOG_KEY = 'rede_points_log'  // 积分明细（流水）记录
const POINTS_INIT_KEY = 'rede_points_init' // 迁移标记：旧版曾把种子 1280 写入本地，升级后统一归零
const SEED_POINTS = 0
const LOG_MAX = 100               // 明细最多保留条数
const STAFF_KEY = 'rede_staff'    // 操作人员（场务/演员）登录态

function todayStr() {
  const d = new Date()
  const p = (n) => (n < 10 ? '0' + n : '' + n)
  return d.getFullYear() + '-' + p(d.getMonth() + 1) + '-' + p(d.getDate())
}

// 首次使用（或旧版升级）写入初始积分；旧版曾把种子 1280 写入本地存储，这里统一归零。
function loadPoints() {
  if (storage.get(POINTS_INIT_KEY, null) !== 'v1') {
    storage.set(POINTS_INIT_KEY, 'v1')
    storage.set(POINTS_KEY, SEED_POINTS)
    return SEED_POINTS
  }
  const v = storage.get(POINTS_KEY, null)
  if (typeof v === 'number') return v
  return SEED_POINTS
}

function loadToday() {
  const v = storage.get(TODAY_KEY, null)
  if (v && typeof v === 'object' && v.date === todayStr()) {
    return typeof v.gain === 'number' ? v.gain : 0
  }
  return 0
}

function loadLog() {
  const v = storage.get(POINT_LOG_KEY, [])
  return Array.isArray(v) ? v : []
}

// 操作人员（场务/演员）登录态：存储 { phone, name, role, roleName }
function loadStaff() {
  const v = storage.get(STAFF_KEY, null)
  return v && typeof v === 'object' ? v : null
}

const state = {
  user: null,        // { openid, nickname, avatar, level, isStaff }
  points: loadPoints(),         // 当前积分余额（持久化）
  isStaff: !!loadStaff(), // 是否为工作人员（场务/演员），随登录态初始化
  staff: loadStaff(),   // 操作人员信息 { phone, name, role, roleName }
  loaded: false,
  todayGain: loadToday(),       // 今日获得的积分（持久化，按日清零）
  todayDate: todayStr(),
  log: loadLog()                // 积分明细流水（持久化，最近 LOG_MAX 条）
}

const subs = new Set()

function notify() {
  subs.forEach((fn) => {
    try { fn(state) } catch (e) { console.error('[store] listener error', e) }
  })
}

// 跨日则重置今日获得
function ensureToday() {
  const t = todayStr()
  if (state.todayDate !== t) {
    state.todayDate = t
    state.todayGain = 0
    storage.set(TODAY_KEY, { date: t, gain: 0 })
  }
}

module.exports = {
  getState() { return state },

  getLog() { return state.log },

  setUser(user) {
    Object.assign(state, user)
    state.loaded = true
    notify()
  },

  // delta 为积分变动量（正=获得，负=消费）；opts 为可选的明细描述：
  //   { type: 'read'|'show'|'postcard'|'redeem'|'adjust', title: '阅读文章', ref: id }
  addPoints(delta, opts) {
    state.points = Math.max(0, state.points + delta)
    // 仅正向奖励累计「今日获得」；消费（负向）不影响今日获得
    if (delta > 0) {
      ensureToday()
      state.todayGain += delta
      storage.set(TODAY_KEY, { date: state.todayDate, gain: state.todayGain })
    }
    storage.set(POINTS_KEY, state.points)
    // 记录积分明细（流水）
    const o = opts || {}
    const entry = {
      id: 'pl' + Date.now() + Math.random().toString(36).slice(2, 6),
      type: o.type || 'adjust',
      title: o.title || (delta > 0 ? '积分奖励' : '积分消费'),
      ref: o.ref || '',
      delta: delta,
      balance: state.points,
      time: Date.now()
    }
    const log = state.log.slice()
    log.unshift(entry)
    if (log.length > LOG_MAX) log.length = LOG_MAX
    state.log = log
    storage.set(POINT_LOG_KEY, log)
    notify()
  },

  setPoints(points) {
    state.points = points
    storage.set(POINTS_KEY, state.points)
    notify()
  },

  // 切换为操作人员（场务/演员）：写入登录态并持久化
  setStaff(info) {
    if (!info || !info.phone) return
    state.staff = Object.assign({}, info)
    state.isStaff = true
    storage.set(STAFF_KEY, state.staff)
    notify()
  },

  // 退出操作人员：清除登录态
  clearStaff() {
    state.staff = null
    state.isStaff = false
    storage.set(STAFF_KEY, null)
    notify()
  },

  subscribe(fn) {
    subs.add(fn)
    // 立即回放一次当前状态
    fn(state)
    return () => subs.delete(fn)
  },

  reset() {
    state.user = null
    state.points = 0
    state.isStaff = false
    state.staff = null
    state.loaded = false
    storage.set(POINTS_KEY, 0)
    storage.set(STAFF_KEY, null)
    storage.set(TODAY_KEY, { date: todayStr(), gain: 0 })
    notify()
  }
}
