// 云函数：aiLineGenerate
// 职责：演员 AI 台词助手 —— 结合角色/情境生成多条备选台词。
//
// 真实接入（腾讯云混元大模型）：
//   - 在云函数控制台「配置 → 环境变量」中配置：
//       TENCENT_SECRET_ID  = 你的腾讯云 SecretId
//       TENCENT_SECRET_KEY = 你的腾讯云 SecretKey
//       TENCENT_REGION     = ap-guangzhou（可选，默认）
//       HUNYUAN_MODEL      = hunyuan-standard（可选，默认）
//   - 未配置或调用失败时，自动回退到内置示例台词（placeholder:true），保证可用。
// 返回字段保持 { role, scene, situation, lines:[{text,tone,reason}], placeholder }。
const cloud = require('wx-server-sdk')
cloud.init({ env: cloud.DYNAMIC_CURRENT_ENV })
const db = cloud.database()
const _ = db.command
const getOpenid = () => cloud.getWXContext()
const ok = (data) => ({ code: 0, data })
const fail = (a, b = 1, extra) => { const code = (typeof a === 'number') ? a : b; const msg = (typeof a === 'number') ? b : a; return extra ? { code, msg, ...extra } : { code, msg } }

const SECRET_ID = process.env.TENCENT_SECRET_ID
const SECRET_KEY = process.env.TENCENT_SECRET_KEY
const REGION = process.env.TENCENT_REGION || 'ap-guangzhou'
const MODEL = process.env.HUNYUAN_MODEL || 'hunyuan-standard'

// 从大模型文本中解析出台词数组（兼容 JSON / 退化分行）
function parseLines(content) {
  if (!content) return null
  let arr = extractJsonArray(content)
  if (Array.isArray(arr) && arr.length) {
    const lines = arr.slice(0, 5).map((it) => ({
      text: String(it.text || it.content || '').trim(),
      tone: String(it.tone || it.emotion || '深情').trim(),
      reason: String(it.reason || it.note || '').trim()
    })).filter((l) => l.text)
    if (lines.length) return lines
  }
  // 退化：按行拆分
  return content.split('\n').map((s) => s.trim()).filter(Boolean).slice(0, 5)
    .map((text) => ({ text, tone: '深情', reason: '' }))
}

function extractJsonArray(s) {
  try {
    const m = s.match(/\[[\s\S]*\]/)
    if (m) return JSON.parse(m[0])
  } catch (e) { /* ignore */ }
  try { return JSON.parse(s) } catch (e) { /* ignore */ }
  return null
}

// 调用腾讯云混元大模型，返回台词数组或 null（走回退）
async function callHunyuan(role, scene, situation) {
  if (!SECRET_ID || !SECRET_KEY) return null
  let hunyuan
  try {
    hunyuan = require('tencentcloud-sdk-nodejs-hunyuan')
  } catch (e) {
    console.warn('[aiLineGenerate] 未安装 tencentcloud-sdk-nodejs-hunyuan，回退示例台词')
    return null
  }
  const client = new hunyuan.hunyuan.v20230901.Client({
    credential: { secretId: SECRET_ID, secretKey: SECRET_KEY },
    region: REGION,
    profile: { httpProfile: { endpoint: 'hunyuan.tencentcloudapi.com' } }
  })
  const prompt =
    `你是沉浸式红色实景演出的编剧助手。请为以下设定生成 3 条可选台词，` +
    `兼顾戏剧张力与史实基调，并标注每条的情绪基调与创作意图。\n` +
    `角色：${role || '主角'}\n场景：${scene || '未知'}\n情境：${situation || ' unspecified'}\n` +
    `请严格以 JSON 数组返回，每条格式：{"text":"台词正文","tone":"情绪基调","reason":"创作意图"}，` +
    `不要输出任何额外说明文字。`
  const resp = await client.ChatCompletions({
    Model: MODEL,
    Messages: [{ Role: 'user', Content: prompt }],
    Stream: false
  })
  const content = resp && resp.Choices && resp.Choices[0] && resp.Choices[0].Message && resp.Choices[0].Message.Content
  return parseLines(content)
}

const FALLBACK_LINES = [
  { text: '同志们，天快亮了。把火把举高点，让后面的人看清路。', tone: '坚定', reason: '在突围情境下稳定军心，呼应「冒险突围」分支' },
  { text: '别出声。听——风里有追兵的马蹄，我们得从这崖缝钻过去。', tone: '警觉', reason: '营造紧张感，推动剧情进入潜行段落' },
  { text: '这封信，我替牺牲的战友带到了。他说，回家替他看一眼麦田。', tone: '深情', reason: '情绪落点，呼应「烽火家书」主题' }
]

exports.main = async (event, context) => {
  const { role, scene, situation } = event

  let lines = null
  let placeholder = false
  try {
    lines = await callHunyuan(role, scene, situation)
  } catch (e) {
    console.warn('[aiLineGenerate] 大模型调用异常，回退示例台词：', e && e.message)
  }
  if (!lines || !lines.length) {
    lines = FALLBACK_LINES
    placeholder = true
  }

  return ok({
    role: role || '主角',
    scene: scene || '',
    situation: situation || '',
    lines,
    placeholder
  })
}

  const { showId, optionId } = event
  if (!OPENID) return fail(401, 'missing openid')
  if (!showId || !optionId) return fail(400, 'missing showId or optionId')

    // 一人一票校验
    const voted = await db
      .collection('vote_records')
      .where({ openid: OPENID, showId })
      .count()
    if (voted.total > 0) {
      return fail(409, 'already voted', { voted: true })
    }

    // 找到该场次的投票题目

    const idx = options.findIndex((o) => o.id === optionId)

    // 原子自增该选项票数
    await db
      .collection('polls')
      .doc(pollId)
      .update({
        data: { [`options.${idx}.votes`]: _.inc(1) }
      })

    // 落库投票记录
    await db.collection('vote_records').add({
      data: {
        openid: OPENID,
        showId,
        optionId,
        created_at: new Date()
      }
    })

    // 返回最新票数
    const fresh = (await db.collection('polls').doc(pollId).get()).data
    return ok({ voted: true, optionId, poll: fresh })
  } catch (e) {
    return fail(500, 'pollVote failed: ' + e.message)

const SENSITIVE = ['毒品', '恐怖', '暴力', '色情', '赌博', '诈骗', '反动', '分裂', '枪支', '炸药']
function hasSensitive(text) {
  return SENSITIVE.some((w) => (text || '').indexOf(w) !== -1)
}

function today() {
  const d = new Date()
  const p = (n) => String(n).padStart(2, '0')
  return `${d.getFullYear()}-${p(d.getMonth() + 1)}-${p(d.getDate())}`
}

// 调用腾讯云 AI 绘画（文生图），返回 base64 或 null（走回退）
async function callTextToImage(prompt) {
  if (!SECRET_ID || !SECRET_KEY) return null
  let aiart
  try {
    aiart = require('tencentcloud-sdk-nodejs-aiart')
  } catch (e) {
    console.warn('[postcardGenerate] 未安装 tencentcloud-sdk-nodejs-aiart，回退 CSS 渲染')
    return null
  }
  const client = new aiart.aiart.v20221229.Client({
    credential: { secretId: SECRET_ID, secretKey: SECRET_KEY },
    region: REGION,
    profile: { httpProfile: { endpoint: 'aiart.tencentcloudapi.com' } }
  })
  const resp = await client.TextToImage({
    Prompt: prompt,
    RspImgType: 'base64'
  })
  return resp && resp.ResultImage ? resp.ResultImage : null
}

// 将 base64 上传云存储并返回可直接 <image> 显示的临时 URL
async function saveImage(base64) {
  const buf = Buffer.from(base64, 'base64')
  const cloudPath = `postcards/${Date.now()}.png`
  const up = await cloud.uploadFile({ cloudPath, fileContent: buf })
  if (!up || !up.fileID) return ''
  const tmp = await cloud.getTempFileURL({ fileList: [up.fileID] })
  const item = tmp && tmp.fileList && tmp.fileList[0]
  return (item && item.tempFileURL) || ''
}

exports.main = async (event, context) => {
  const { OPENID } = getOpenid()
  const { sceneId, quote, customText, orient, arrange, font, position } = event
  const scene = SCENES[sceneId] || SCENES.sc1

  // 自定义金句敏感词拦截（与前端一致）
  if (customText && hasSensitive(customText)) {
    return fail('内容包含不适当的词汇，请修改后再生成', 400)
  }

  // 自定义金句为空时返回空字符串（原图不带名言）；仅当用户确实输入才带名言。
  // 不再随机兜底 QUOTES，否则导出图总会带一句预设名言。
  const finalQuote = (customText || '').trim()
  const ORIENTS = ['land', 'port']
  const ARRANGES = ['h', 'v']
  const FONTS = ['maoti', 'shoujin', 'liu', 'yan']
  const POSITIONS = ['top', 'bottom', 'left', 'right', 'tl', 'tr', 'bl', 'br', 'center']
  const finalOrient = ORIENTS.indexOf(orient) !== -1 ? orient : 'land'
  const finalArrange = ARRANGES.indexOf(arrange) !== -1 ? arrange : 'h'
  const finalFont = FONTS.indexOf(font) !== -1 ? font : 'maoti'
  const finalPosition = POSITIONS.indexOf(position) !== -1 ? position : 'center'

  let imageUrl = ''
  if (SECRET_ID && SECRET_KEY) {
    try {
      const prompt =
        `艺术海报风格插画，主题「${scene.name}」：${scene.desc}。` +
        `电影感光照，庄重而温暖的革命历史氛围，构图留白充足，无人物特写，不要任何文字。` +
        `配色参考：${scene.colors.join('、')}。`
      const b64 = await callTextToImage(prompt)
      if (b64) imageUrl = await saveImage(b64)
    } catch (e) {
      console.warn('[postcardGenerate] 文生图失败，回退 CSS 渲染：', e && e.message)
    }
  }
  const placeholder = !imageUrl

  const result = {
    id: 'p' + Date.now(),
    scene: scene.name,
    gradient: scene.gradient,
    colors: scene.colors,
    quote: finalQuote,
    date: today(),
    orient: finalOrient,
    arrange: finalArrange,
    font: finalFont,
    position: finalPosition,
    imageUrl,
    placeholder
  }

  // 登录态下可落库（集合不存在时云开发自动创建）
  if (OPENID) {
    try {
      await db.collection('postcards').add({
        data: {
          openid: OPENID,
          scene: scene.name,
          gradient: scene.gradient,
          colors: scene.colors,
          quote: finalQuote,
          date: result.date,
          orient: finalOrient,
          arrange: finalArrange,
          font: finalFont,
          position: finalPosition,
          imageUrl,
          created_at: new Date()
        }
      })
    } catch (e) { /* 落库失败不影响返回 */ }
  }

  return ok(result)
}

// 云函数：showReview
// 职责：演出回顾（场次信息 + 我的投票 + 情绪分布 + 一句金句）。
// 读取集合：shows / vote_records / emotion_feedbacks / quotes
const cloud = require('wx-server-sdk')
cloud.init({ env: cloud.DYNAMIC_CURRENT_ENV })
const db = cloud.database()
const _ = db.command
const $ = db.command.aggregate
const getOpenid = () => cloud.getWXContext()
const ok = (data) => ({ code: 0, data })
const fail = (a, b = 1, extra) => { const code = (typeof a === 'number') ? a : b; const msg = (typeof a === 'number') ? b : a; return extra ? { code, msg, ...extra } : { code, msg } }

// 情绪类型（与 getShowRoom / 前端 re-tag 配色一致）
const EMOTIONS = [
  { id: 'e1', label: '感动' },
  { id: 'e2', label: '震撼' },
  { id: 'e3', label: '激昂' },
  { id: 'e4', label: '沉思' }
]
// 无数据时的默认分布（避免前端空白）
const DEFAULT_EMO = [
  { label: '感动', pct: 42 },
  { label: '震撼', pct: 31 },
  { label: '激昂', pct: 19 },
  { label: '沉思', pct: 8 }
]
const DEFAULT_QUOTE = '为有牺牲多壮志，敢教日月换新天。'

exports.main = async (event, context) => {
  const { OPENID } = getOpenid()
  if (!OPENID) return fail('missing openid', 401)

  const showId = event.showId
  if (!showId) return fail('missing showId', 400)

  try {
    // 场次
    const sRes = await db.collection('shows').where({ id: showId }).get()
    const show = sRes.data[0] || null

    // 我的投票（该用户在该场次的最新一票）
    let myVote = ''
    const vRes = await db
      .collection('vote_records')
      .where({ openid: OPENID, showId })
      .orderBy('created_at', 'desc')
      .limit(1)
      .get()
    if (vRes.data[0]) {
      const v = vRes.data[0]
      myVote = v.optionLabel || v.optionId || ''
    }

    // 情绪分布（按 type 聚合 → 百分比）
    let emotionDist = DEFAULT_EMO
    const agg = await db
      .collection('emotion_feedbacks')
      .aggregate()
      .match({ showId })
      .group({ _id: '$type', count: $.sum(1) })
      .end()
    if (agg.list && agg.list.length) {
      const total = agg.list.reduce((s, x) => s + x.count, 0)
      emotionDist = EMOTIONS.map((e) => {
        const hit = agg.list.find((x) => x._id === e.id)
        const c = hit ? hit.count : 0
        return { label: e.label, pct: total ? Math.round((c / total) * 100) : 0 }
      })
    }

    // 一句金句（随机取一条，兼容字符串/对象存储）
    let quote = DEFAULT_QUOTE
    const qAgg = await db.collection('quotes').aggregate().sample({ size: 1 }).end()
    if (qAgg.list && qAgg.list[0]) {
      const q = qAgg.list[0]
      quote = typeof q === 'string' ? q : (q.text || DEFAULT_QUOTE)
    }

    return ok({ show, myVote, emotionDist, quote })
  } catch (e) {
    return fail('showReview failed: ' + e.message, 500)
  }
}

// components/pc-canvas/pc-canvas.js
// 复用 services/pcLayout.js 的 paintPostcard，将明信片以 Canvas 绘制。
// 展示统一为竖版(2:3)：横版内容在 canvas 内部顺时针旋转 90°（与导出一致）。
// <canvas> 元素始终 100% 填满父容器，无裁剪问题。
const { paintPostcard, decorateCard } = require('../../services/pcLayout.js')

// 参考分辨率：统一竖版 600×900（与 .pc-frame aspect-ratio: 2/3 对应）
const REF_W = 600
const REF_H = 900

function getDpr() {
  try {
    if (wx.getWindowInfo) return wx.getWindowInfo().pixelRatio || 2
    return wx.getSystemInfoSync().pixelRatio || 2
  } catch (e) { return 2 }
}

Component({
  properties: {
    card: { type: Object, value: null, observer() { this._render() } },
    // 覆盖长边上限 px（未乘 dpr）。预览传 900 高清；默认缩略图 480 控内存。
    res: { type: Number, value: 0 }
  },

  lifetimes: {
    attached() { this._canvas = null },
    ready() { this._query() }
  },

  pageLifetimes: {
    show() { this._render() } // 字体加载后重绘用上真·书法体
  },

  methods: {
    _query() {
      wx.createSelectorQuery().in(this).select('.cv')
        .fields({ node: true, size: true })
        .exec((res) => {
          if (res && res[0] && res[0].node) {
            this._canvas = res[0].node
            this._render()
          }
        })
    },

    _render() {
      const canvas = this._canvas
      const card = this.data.card
      if (!canvas || !card) return

      const orient = card.orient || 'land'
      // 长边上限：按竖版 900 为基准
      const maxLong = (this.data.res && this.data.res > 0) ? this.data.res : 480
      const scale = Math.min(maxLong / REF_W, maxLong / REF_H)

      const dpr = getDpr()
      const W = Math.round(REF_W * scale)
      const H = Math.round(REF_H * scale)
      canvas.width = W * dpr
      canvas.height = H * dpr

      const ctx = canvas.getContext('2d')
      ctx.setTransform(1, 0, 0, 1, 0, 0)
      ctx.clearRect(0, 0, canvas.width, canvas.height)
      ctx.scale(dpr * scale, dpr * scale)

      const c = decorateCard(card)
      if (orient === 'land') {
        // 横版：在 600×900 竖版画布内，translate(600,0)+rotate(π/2) 把 3:2 组合旋转填满
        ctx.translate(REF_W, 0)
        ctx.rotate(Math.PI / 2)
        paintPostcard(ctx, 900, 600, c)
      } else {
        paintPostcard(ctx, REF_W, REF_H, c)
      }
    }
  }
})

// packages/create/pages/generate/generate.js
const api = require('../../../../services/api.js')
const postcards = require('../../../../services/postcards.js')
const { containsSensitive } = require('../../services/sensitive.js')
const { FONTS, loadFonts } = require('../../../../services/fonts.js')
const { decorateCard, paintPostcard, renderPostcard } = require('../../../../services/pcLayout.js')
const { clampQuote, quoteCount, MAX_QUOTE } = require('../../services/quote.js')
const store = require('../../../../store.js')
const rewards = require('../../../../services/rewards.js')

// 字体风格 → 仅暴露修饰词（不展示具体字体名）
const fontList = Object.keys(FONTS).map((key) => ({
  key,
  name: FONTS[key].adjective,
  desc: FONTS[key].desc
}))

// 位置中文名（用于展示）
const POSITION_NAMES = {
  top: '上', center: '中', bottom: '下',
  left: '左', right: '右'
}

// 按「排列」限定可选位置（横排：上/中/下；竖排：左/中/右）
const POSITIONS_BY_MODE = {
  h: ['top', 'center', 'bottom'],
  v: ['left', 'center', 'right']
}

// 各排列的默认位置（切换排列时回退到该默认，而非首个，体验更自然）
const POSITION_DEFAULT = { h: 'center', v: 'right' }

// 根据排列返回当前可选位置列表（[{key,name}]）
function getPositionList(orient, arrange) {
  const keys = POSITIONS_BY_MODE[arrange || 'h'] || POSITIONS_BY_MODE['h']
  return keys.map((k) => ({ key: k, name: POSITION_NAMES[k] || k }))
}

// 当前位置是否仍在可选范围内，否则回退到该排列的默认位置
function normalPosition(orient, arrange, position) {
  const list = getPositionList(orient, arrange)
  if (list.some((p) => p.key === position)) return position
  return POSITION_DEFAULT[arrange] || (list[0] ? list[0].key : 'top')
}

Page({
  data: {
    scenes: [],
    result: null,
    generating: false,
    useImage: false,
    genHint: '',
    // 创作面板状态
    selectedId: '',
    customText: '',
    maxQuote: MAX_QUOTE,   // 金句有效字数上限（不含换行，标点计入）
    quoteCount: 0,         // 当前有效字数（实时计数展示用）
    orient: 'land',   // land 横版 / port 竖版（匹配图片朝向，按实物明信片比例）
    arrange: 'h',     // h 横排 / v 竖排
    position: 'center', // 名言 9 宫格位置
    font: 'maoti',    // 字体风格 key（UI 仅显示修饰词）
    draft: {},
    presets: [
      '星星之火，可以燎原。',
      '为有牺牲多壮志，敢教日月换新天。',
      '江山就是人民，人民就是江山。',
      '一代人有一代人的长征。'
    ],
    orientList: [
      { key: 'land', name: '横版', desc: '适配横向图' },
      { key: 'port', name: '竖版', desc: '适配竖向图' }
    ],
    arrangeList: [
      { key: 'h', name: '横排', desc: '从左向右' },
      { key: 'v', name: '竖排', desc: '从上向下' }
    ],
    positionList: getPositionList('land', 'h'),
    fontList
  },

  tempPath: '',

  onLoad(options) {
    loadFonts()
    api.getScenes().then((scenes) => {
      this.setData({
        scenes,
        selectedId: options.sceneId || (scenes[0] && scenes[0].id) || ''
      }, () => { this.updateDraft(); this.updateGenHint() })
    })
  },

  onSelectScene(e) {
    this.setData({ selectedId: e.currentTarget.dataset.id }, () => this.updateDraft())
  },

  onInputQuote(e) {
    const { text, count } = clampQuote(e.detail.value)
    this.setData({ customText: text, quoteCount: count }, () => this.updateDraft())
  },

  onPickPreset(e) {
    const { text, count } = clampQuote(e.currentTarget.dataset.q)
    this.setData({ customText: text, quoteCount: count }, () => this.updateDraft())
  },

  onSelectOrient(e) {
    const orient = e.currentTarget.dataset.key
    this.setData({
      orient,
      positionList: getPositionList(orient, this.data.arrange),
      position: normalPosition(orient, this.data.arrange, this.data.position)
    }, () => this.updateDraft())
  },

  onSelectArrange(e) {
    const arrange = e.currentTarget.dataset.key
    this.setData({
      arrange,
      positionList: getPositionList(this.data.orient, arrange),
      position: normalPosition(this.data.orient, arrange, this.data.position)
    }, () => this.updateDraft())
  },

  onSelectPosition(e) {
    this.setData({ position: e.currentTarget.dataset.key }, () => this.updateDraft())
  },

  onSelectFont(e) {
    this.setData({ font: e.currentTarget.dataset.key }, () => this.updateDraft())
  },

  // 根据当前选择实时构建预览草稿（含排版引擎计算的名言分行/字号）
  updateDraft() {
    const scenes = this.data.scenes
    const scene = scenes.find((s) => s.id === this.data.selectedId) || scenes[0] || {}
    const text = (this.data.customText && this.data.customText.trim()) || ""
    const base = {
      gradient: scene.gradient || 'linear-gradient(135deg,#C8161D,#7A0A10)',
      colors: scene.colors || ['#C8161D', '#7A0A10'],
      scene: scene.name || '红色旧址',
      quote: text,
      orient: this.data.orient,
      arrange: this.data.arrange,
      position: this.data.position,
      font: this.data.font,
      date: postcards.today()
    }
    this.setData({ draft: decorateCard(base) })
  },

  // 刷新「明信片计费」提示：前 5 次免费，之后每次 -1 积分
  updateGenHint() {
    const st = rewards.postcardState()
    const hint = st.free
      ? ('前 5 次免费 · 当前第 ' + (st.count + 1) + ' 次')
      : '本次生成消耗 1 积分'
    this.setData({ genHint: hint })
  },

  onGenerate() {
    const { selectedId, customText, orient, arrange, position, font } = this.data
    if (!selectedId) {
      wx.showToast({ title: '请先选择场景', icon: 'none' })
      return
    }
    if (containsSensitive(customText)) {
      wx.showModal({
        title: '内容需调整',
        content: '明信片文字包含不适当的词汇，请修改后再生成。',
        showCancel: false,
        confirmText: '我知道了'
      })
      return
    }
    // 明信片计费：前 5 次免费，之后每次 -1 积分；积分不足则拦截
    const st = rewards.postcardState()
    if (!st.free && store.getState().points < 1) {
      wx.showToast({ title: '积分不足，无法生成', icon: 'none' })
      return
    }
    this.setData({ generating: true, result: null, useImage: false })
    const opts = { orient, arrange, position, font }
    api.generatePostcard(selectedId, customText, opts).then((res) => {
      const scene = this.data.scenes.find((s) => s.id === selectedId) || {}
      const finalRec = Object.assign({}, res, {
        scene: scene.name || res.scene,
        gradient: scene.gradient || res.gradient,
        colors: scene.colors || res.colors,
        quote: (customText && customText.trim()) || '',
        orient, arrange, position, font,
        date: postcards.today()
      })
      // 排版引擎补全渲染字段
      const decorated = decorateCard(finalRec)
      // 落库：保证「已存入我的作品」真实可信
      postcards.add(decorated)
      // 计费提交：计数 +1；超过免费次数则扣 1 积分
      const c = rewards.commitPostcard()
      this.updateGenHint()
      const tip = c.free ? '已存入（免费）' : '已存入（-1 积分）'
      wx.showToast({ title: tip, icon: 'success' })
      this.setData({ result: decorated, generating: false, useImage: !!res.imageUrl }, () => {
        if (res.imageUrl) this.loadImage(res.imageUrl)
      })
    }).catch((err) => {
      this.setData({ generating: false })
      wx.showToast({ title: (err && err.message) || '生成失败', icon: 'none' })
    })
  },

  loadImage(url) {
    wx.downloadFile({
      url,
      success: (e) => {
        if (e.statusCode === 200 && e.tempFilePath) {
          this.tempPath = e.tempFilePath
          this.setData({ useImage: true })
        }
      },
      fail: () => {}
    })
  },

  // 离屏画布绘制明信片，导出临时文件用于保存/打印（返回 Promise<tempPath>）
  // 导出统一为竖版(2:3)，横版内容顺时针旋转 90°（与展示态一致）
  drawPostcard() {
    const r = this.data.result
    if (!r) return Promise.resolve('')
    return new Promise((resolve) => {
      wx.createSelectorQuery().in(this)
        .select('#pcCanvas')
        .fields({ node: true, size: true })
        .exec((res) => {
          if (!res || !res[0] || !res[0].node) return resolve('')
          const canvas = res[0].node
          renderPostcard(canvas, r)
          wx.canvasToTempFilePath({
            canvas,
            success: (e) => { this.tempPath = e.tempFilePath; resolve(e.tempFilePath) },
            fail: () => resolve('')
          })
        })
    })
  },

  onSave() {
    const save = (path) => {
      if (!path) { wx.showToast({ title: '稍候再试', icon: 'none' }); return }
      wx.saveImageToPhotosAlbum({
        filePath: path,
        success: () => wx.showToast({ title: '已保存相册', icon: 'success' }),
        fail: (err) => {
          if (/auth|deny/i.test(err.errMsg || '')) {
            wx.showModal({
              title: '需要相册权限',
              content: '请在设置中开启「保存到相册」权限',
              confirmText: '去设置',
              success: (m) => { if (m.confirm) wx.openSetting() }
            })
          } else {
            wx.showToast({ title: '保存失败', icon: 'none' })
          }
        }
      })
    }
    if (this.tempPath) { save(this.tempPath); return }
    this.drawPostcard().then((path) => save(path))
  },

  // 打印：绘制高清图后用系统图片查看器（可长按保存 / 打印）
  onPrint() {
    const rec = this.data.result
    if (!rec) return
    wx.showToast({ title: '长按图片可保存或打印', icon: 'none' })
    const open = (path) => { if (path) wx.previewImage({ urls: [path], current: path }) }
    if (rec.imageUrl) { open(rec.imageUrl); return }
    this.drawPostcard().then(open)
  },

  onAgain() {
    this.tempPath = ''
    this.setData({ result: null, useImage: false, customText: '', quoteCount: 0 }, () => { this.updateDraft(); this.updateGenHint() })
  },

  onShareAppMessage() {
    const r = this.data.result
    return {
      title: r ? ('我在红韵智演创作了「' + r.scene + '」明信片') : '红韵智演 · 创作你的专属红色明信片',
      path: '/pages/create/create',
      imageUrl: this.tempPath || (r && r.imageUrl) || ''
    }
  }
})

// services/quote.js —— 金句输入框约束
// 规则：有效字数 = 去掉换行后的字符数（标点也算）；上限 20；超出则截断并保留换行结构。
const MAX_QUOTE = 20

// 对输入做约束：保留换行，但非换行字符最多保留 limit 个（默认 20）
// 返回 { text: 约束后的文本, count: 有效字数(不含换行) }
function clampQuote(v, limit) {
  const max = (typeof limit === 'number' && limit > 0) ? limit : MAX_QUOTE
  const s = (v == null) ? '' : String(v)
  let count = 0
  let out = ''
  for (const ch of s) {
    if (ch === '\n' || ch === '\r') { out += ch; continue } // 换行不计入、也不截断
    if (count < max) { out += ch; count++ }
  }
  return { text: out, count }
}

// 仅统计有效字数（不含换行），用于实时计数展示
function quoteCount(v) {
  return String(v == null ? '' : v).replace(/\r?\n/g, '').length
}

module.exports = { clampQuote, quoteCount, MAX_QUOTE }

// services/sensitive.js —— 轻量敏感词过滤（前端演示用）
// 生产环境应以服务端审核为准；此处仅做客户端基础拦截，避免明显违规内容进入生成流程。
const WORDS = ['毒品', '恐怖', '暴力', '色情', '赌博', '诈骗', '反动', '分裂', '枪支', '炸药']

function containsSensitive(text) {
  if (!text) return false
  const t = String(text)
  return WORDS.some((w) => t.indexOf(w) !== -1)
}

module.exports = { containsSensitive, WORDS }

// packages/show/pages/room/room.js
const api = require('../../../../services/api.js')
const rewards = require('../../../../services/rewards.js')

function computePct(options) {
  const total = options.reduce((s, o) => s + o.votes, 0) || 1
  return options.map((o) => Object.assign({}, o, { pct: Math.round((o.votes / total) * 100) }))
}

Page({
  data: {
    showId: '',
    show: { title: '', segment: '', online: 0, gradient: '', status: '', pollOpen: false },
    poll: { question: '', options: [] },
    emotions: [],
    branch: { text: '', current: '', locked: false },
    notices: [],
    messages: [],
    voted: '',
    selectedEmotion: '',
    inputValue: ''
  },

  timer: null,

  onLoad(options) {
    const showId = options.id || 's1'
    this.setData({ showId })
    // 参加一次演出 +5 积分（同一场只加一次）
    if (rewards.awardShow(showId)) {
      wx.showToast({ title: '参与演出 +5 积分', icon: 'none' })
    }
    api.getShowRoom(showId).then((d) => {
      this.setData({
        show: d.show,
        poll: Object.assign({}, d.poll, { options: computePct(d.poll.options) }),
        emotions: d.emotions,
        branch: d.branch,
        notices: d.notices
      })
      // 模拟实时：仅直播且投票开启时轮询结果
      if (d.show.pollOpen) {
        this.timer = setInterval(() => this.tick(), 2600)
      }
    })
  },

  onUnload() {
    if (this.timer) clearInterval(this.timer)
  },

  tick() {
    api.pollResult(this.data.showId).then((opts) => {
      this.setData({ 'poll.options': computePct(opts) })
    })
  },

  onVote(e) {
    if (this.data.voted) return
    const id = e.currentTarget.dataset.id
    api.submitVote(this.data.showId, id).then(() => {
      const opts = this.data.poll.options.map((o) =>
        o.id === id ? Object.assign({}, o, { votes: o.votes + 1 }) : o
      )
      this.setData({ voted: id, 'poll.options': computePct(opts) })
      wx.showToast({ title: '投票成功', icon: 'success' })
    })
  },

  onEmotion(e) {
    const id = e.currentTarget.dataset.id
    this.setData({ selectedEmotion: id })
    api.submitEmotion(this.data.showId, id) //  fire-and-forget
  },

  onInput(e) {
    this.setData({ inputValue: e.detail.value })
  },

  onSend() {
    const text = (this.data.inputValue || '').trim()
    if (!text) return
    const msg = { id: 'm' + Date.now(), text, time: this.now() }
    this.setData({ messages: this.data.messages.concat(msg), inputValue: '' })
  },

  goGenerate() {
    wx.navigateTo({ url: '/packages/create/pages/generate/generate' })
  },

  now() {
    const d = new Date()
    return `${d.getHours()}:${String(d.getMinutes()).padStart(2, '0')}`
  }
})

// 创作页「我的作品」最多展示条数
const MAX_SHOW = 4

Page({
  data: {
    loading: true,
    postcards: [],
    hasWorks: false,
    preview: null
  },

  onShow() {
    loadFonts()
    this.load()
    // 回到本页时：若预览仍打开则保持 tabBar 隐藏（如打印系统看图器关闭会触发 onShow），
    // 否则恢复显示；同时高亮当前 tab。
    if (typeof this.getTabBar === 'function' && this.getTabBar()) {
      this.getTabBar().setData({ selected: 2, hidden: !!this.data.preview })
    }
  },

  load() {
    this.setData({ loading: true })
    api.getPostcards()
      .then((list) => {
        const decorated = list.map((p) => decorateCard(p))
        this.setData({
          postcards: decorated.slice(0, MAX_SHOW),
          hasWorks: list.length > 0,
          loading: false
        })
      })
      .catch(() => this.setData({ loading: false }))
  },

  // 「创作红色记忆」卡片入口：进入生成页（场景选择留在生成页内）
  onStartCreate() {
    wx.navigateTo({ url: '/packages/create/pages/generate/generate' })
  },

  // 「全部」：跳转我的作品列表页
  onAll() {
    wx.navigateTo({ url: '/pages/works/works' })
  },

  // 画廊点击 → 预览
  onPreview(e) {
    const id = e.currentTarget.dataset.id
    const item = this.data.postcards.find((p) => p.id === id)
    if (!item) return
    // 打开预览前先隐藏自定义 tabBar（组件 hidden 状态），否则 tabBar 会浮在遮罩之上仍可点击
    if (typeof this.getTabBar === 'function' && this.getTabBar()) {
      this.getTabBar().setData({ hidden: true })
    }
    this.setData({ preview: item })
  },

  onClosePreview() {
    this.setData({ preview: null })
    if (typeof this.getTabBar === 'function' && this.getTabBar()) {
      this.getTabBar().setData({ hidden: false })
    }
  },

  // 画廊长按删除作品
  onDeleteWork(e) {
    const id = e.currentTarget.dataset.id
    if (!id) return
    wx.showModal({
      title: '删除作品',
      content: '确定删除这张明信片吗？删除后不可恢复。',
      confirmText: '删除',
      confirmColor: '#C8161D',
      success: (m) => {
        if (m.confirm) {
          postcards.remove(id)
          if (this.data.preview && this.data.preview.id === id) {
            this.setData({ preview: null })
            // 预览中被删除，关闭预览的同时恢复 tabBar
            if (typeof this.getTabBar === 'function' && this.getTabBar()) {
              this.getTabBar().setData({ hidden: false })
            }
          }
          this.load()
          wx.showToast({ title: '已删除', icon: 'success' })
        }
      }
    })
  },

  // 从历史记录现绘明信片（无 imageUrl 时），返回 Promise<tempPath>
  // 导出统一为竖版(2:3)，横版内容顺时针旋转 90°（与展示态一致）
  drawRecord(rec) {
    return new Promise((resolve) => {
      if (!rec) return resolve('')
      wx.createSelectorQuery().in(this)
        .select('#pcCanvas')
        .fields({ node: true, size: true })
        .exec((res) => {
          if (!res || !res[0] || !res[0].node) return resolve('')
          const canvas = res[0].node
          renderPostcard(canvas, rec)
          wx.canvasToTempFilePath({
            canvas,
            success: (e) => resolve(e.tempFilePath),
            fail: () => resolve('')
          })
        })
    })
  },

  // 预览层「保存相册」
  onSaveFromPreview() {
    const rec = this.data.preview
    if (!rec) return
    const save = (path) => {
      if (!path) { wx.showToast({ title: '稍候再试', icon: 'none' }); return }
      wx.saveImageToPhotosAlbum({
        filePath: path,
        success: () => wx.showToast({ title: '已保存相册', icon: 'success' }),
        fail: (err) => {
          if (/auth|deny/i.test(err.errMsg || '')) {
            wx.showModal({
              title: '需要相册权限',
              content: '请在设置中开启「保存到相册」权限',
              confirmText: '去设置',
              success: (m) => { if (m.confirm) wx.openSetting() }
            })
          } else {
            wx.showToast({ title: '保存失败', icon: 'none' })
          }
        }
      })
    }
    if (rec.imageUrl) { save(rec.imageUrl); return }
    this.drawRecord(rec).then((path) => save(path))
  },

  // 打印：绘制高清图后用系统图片查看器（可长按保存 / 打印）
  onPrint() {
    const rec = this.data.preview
    if (!rec) return
    wx.showToast({ title: '长按图片可保存或打印', icon: 'none' })
    const open = (path) => { if (path) wx.previewImage({ urls: [path], current: path }) }
    if (rec.imageUrl) { open(rec.imageUrl); return }
    this.drawRecord(rec).then(open)
  },

  // 转发给好友（由预览层 <button open-type="share"> 触发）
  onShareAppMessage() {
    const p = this.data.preview
    return {
      title: p ? ('我在红韵智演创作了「' + p.scene + '」明信片') : '红韵智演 · 创作你的专属红色明信片',
      path: '/pages/create/create',
      imageUrl: p && p.imageUrl ? p.imageUrl : ''
    }
  }
})

// pages/favorites/favorites.js —— 阅读收藏：已收藏篇章列表（本地收藏）
const res = require('../../services/resources.js')
const favorites = require('../../services/favorites.js')

const CAT_NAME = res.CATEGORIES.reduce((m, c) => { m[c.key] = c.name; return m }, {})
const FAV_SHOW = 200 // 最多展示条数

const SWIPE_BTN = 72        // 删除按钮宽度（px）
const SWIPE_OPEN = -SWIPE_BTN   // 完全展开时的偏移
const SWIPE_THRESHOLD = 36  // 位移超过该阈值则自动展开（px）

Page({
  data: { list: [], count: 0 },

  // 拖拽过程临时状态（不放入 data，避免无谓渲染）
  _dragId: null,
  _startX: 0,
  _startOffset: 0,

  onShow() { this.refresh() },

  refresh() {
    const ids = favorites.getList()
    const list = ids
      .map((id) => res.cards.find((c) => c.id === id))
      .filter(Boolean)
      .slice(0, FAV_SHOW)
      .map((c) => ({
        id: c.id,
        title: c.title,
        subtitle: c.subtitle,
        catName: CAT_NAME[c.cat] || '',
        gradient: c.gradient,
        offset: 0,   // 右滑偏移（px，负数表示露出删除键）
        anim: false  // 偏移变化时是否启用过渡动画
      }))
    this.setData({ list, count: ids.length })
  },

  idxOf(id) {
    return this.data.list.findIndex((x) => x.id === id)
  },

  onRead(e) {
    const id = e.currentTarget.dataset.id
    const i = this.idxOf(id)
    const item = this.data.list[i]
    // 已滑开（露出删除键）时，点击内容先收起而非跳转
    if (item && item.offset) {
      this.setData({ [`list[${i}].offset`]: 0, [`list[${i}].anim`]: true })
      return
    }
    wx.navigateTo({ url: '/pages/resource/detail/detail?id=' + id })
  },

  // —— 右滑删除手势 ——
  onSwipeStart(e) {
    const id = e.currentTarget.dataset.id
    const i = this.idxOf(id)
    if (i < 0) return
    this._dragId = id
    this._startX = e.touches[0].clientX
    this._startOffset = this.data.list[i].offset || 0
    // 拖拽中关闭过渡动画，并把其它已展开的项收起
    const updates = {}
    this.data.list.forEach((it, idx) => {
      if (it.id !== id && it.offset) updates[`list[${idx}].offset`] = 0
    })
    updates[`list[${i}].anim`] = false
    this.setData(updates)
  },

  onSwipeMove(e) {
    if (!this._dragId) return
    const i = this.idxOf(this._dragId)
    if (i < 0) return
    let offset = this._startOffset + (e.touches[0].clientX - this._startX)
    if (offset > 0) offset = 0
    if (offset < SWIPE_OPEN) offset = SWIPE_OPEN
    this.setData({ [`list[${i}].offset`]: offset })
  },

  onSwipeEnd(e) {
    if (!this._dragId) return
    const i = this.idxOf(this._dragId)
    this._dragId = null
    if (i < 0) return
    const offset = this.data.list[i].offset || 0
    const target = offset <= -SWIPE_THRESHOLD ? SWIPE_OPEN : 0
    this.setData({ [`list[${i}].offset`]: target, [`list[${i}].anim`]: true })
  },

  // 点击删除键：移除单条收藏
  onDelete(e) {
    const id = e.currentTarget.dataset.id
    favorites.remove(id)
    this.refresh()
  },

  onClear() {
    wx.showModal({
      title: '清空收藏',
      content: '确定清空全部收藏？',
      success: (r) => { if (r.confirm) { favorites.clear(); this.refresh() } }
    })
  }
})

// pages/home/home.js
const api = require('../../services/api.js')
const store = require('../../store.js')
const res = require('../../services/resources.js')
const readHistory = require('../../services/readHistory.js')

const HOME_MAX = 3 // 首页随机推荐 3 条（从整个资料库随机挑选）
const CAT_NAME = res.CATEGORIES.reduce((m, c) => { m[c.key] = c.name; return m }, {})
const READ_SHOW = 3 // 红色足迹仅展示最近 3 篇

Page({
  data: {
    loading: true,
    home: { greeting: '你好', points: 0, todayGain: 0, recommend: [], reading: [], liveShow: null },
    nickname: '旅人',
    avatarText: '红',
    hasMore: false
  },

  unsub: null,

  onLoad() {
    this.unsub = store.subscribe((s) => {
      if (s.user) this.setData({ nickname: s.user.nickname, avatarText: s.user.avatarText })
      // 跨页积分变动后回到首页即时反映
      this.setData({ 'home.points': s.points, 'home.todayGain': s.todayGain })
    })
    this.load()
  },

  onShow() {
    // 红色足迹始终反映最新已读状态
    this.buildReading()
    // 由推荐卡片进入详情并返回时，仅移除刚读的那篇并在尾部补一篇（不整体刷新）
    if (this._pendingReadId) {
      const id = this._pendingReadId
      this._pendingReadId = null
      this.removeAndAppend(id)
    }
    // 同步自定义 tabBar 选中态
    if (typeof this.getTabBar === 'function' && this.getTabBar()) {
      this.getTabBar().setData({ selected: 0 })
    }
  },

  onUnload() {
    if (this.unsub) this.unsub()
  },

  // 从整个资料库随机抽取 n 条（Fisher–Yates 洗牌）
  pickRandom(list, n) {
    const arr = list.slice()
    for (let i = arr.length - 1; i > 0; i--) {
      const j = Math.floor(Math.random() * (i + 1))
      const t = arr[i]; arr[i] = arr[j]; arr[j] = t
    }
    return arr.slice(0, n)
  },

  // 为你推荐：优先推送未读文章；未读不足 HOME_MAX 时以已读补足；全部读完则继续随机
  buildRecommend() {
    const readIds = new Set(readHistory.getList())
    const unread = res.cards.filter((c) => !readIds.has(c.id))
    const read = res.cards.filter((c) => readIds.has(c.id))
    let pool = this.pickRandom(unread, HOME_MAX)
    if (pool.length < HOME_MAX) {
      pool = pool.concat(this.pickRandom(read, HOME_MAX - pool.length))
    }
    const recommend = pool.map((c) => Object.assign({}, c, { read: readIds.has(c.id) }))
    this.setData({ 'home.recommend': recommend, hasMore: res.cards.length > HOME_MAX })
  },

  // 红色足迹：从本地阅读历史读取，仅展示最近 3 篇；与「我的」页共用 readHistory 数据源
  buildReading() {
    const list = readHistory.getEntries()
      .map((e) => {
        const c = res.cards.find((x) => x.id === e.id)
        if (!c) return null
        return {
          id: c.id,
          title: c.title,
          catName: CAT_NAME[c.cat] || '',
          gradient: c.gradient,
          readTime: readHistory.formatDate(e.time)
        }
      })
      .filter(Boolean)
      .slice(0, READ_SHOW)
    this.setData({ 'home.reading': list })
  },

  load() {
    this.setData({ loading: true })
    api.getHomeData()
      .then((d) => {
        this.setData({
          home: Object.assign({}, this.data.home, {
            greeting: d.greeting,
            points: d.points,
            todayGain: d.todayGain,
            liveShow: d.liveShow
          }),
          loading: false
        })
        this.buildReading()
        this.buildRecommend()
        wx.stopPullDownRefresh()
      })
      .catch(() => {
        // 云端异常时仍保证资料卡片可见
        this.setData({ loading: false })
        this.buildReading()
        this.buildRecommend()
        wx.stopPullDownRefresh()
      })
  },

  onPullDownRefresh() {
    this.load()
  },

  // 进入「更多」资料列表页（列表页内再提供分类筛选）
  onMore() {
    wx.navigateTo({ url: '/pages/resource/list/list' })
  },

  // 红色足迹「全部」按钮：跳转我的阅读列表页
  onAllReads() {
    wx.navigateTo({ url: '/pages/read/read' })
  },

  // 点击推荐卡片：记录待移除 id，进入详情页（返回时不整体刷新，仅移除该篇并尾部补一篇）
  onRead(e) {
    const id = e.currentTarget.dataset.id
    this._pendingReadId = id
    wx.navigateTo({ url: `/pages/resource/detail/detail?id=${id}` })
  },

  // 由推荐卡片返回时调用：移除刚刚阅读过的那篇，并在列表尾部补充一篇（优先未读）
  removeAndAppend(justReadId) {
    const readIds = new Set(readHistory.getList())
    let list = (this.data.home.recommend || []).slice()
    // 仅移除刚刚阅读过的那篇（其余保持不变，不整体刷新）
    list = list.filter((c) => c.id !== justReadId)
    const existIds = new Set(list.map((c) => c.id))
    // 从「未读且不在当前列表」中补一篇；不足则从已读补
    const unread = res.cards.filter((c) => !readIds.has(c.id) && !existIds.has(c.id))
    const read = res.cards.filter((c) => readIds.has(c.id) && !existIds.has(c.id))
    let pick = this.pickRandom(unread, 1)
    if (pick.length < 1) pick = this.pickRandom(read, 1)
    const appended = pick.map((c) => Object.assign({}, c, { read: readIds.has(c.id) }))
    list = list.concat(appended)
    this.setData({ 'home.recommend': list, hasMore: res.cards.length > list.length })
  },

  goRoom(e) {
    const id = e.currentTarget.dataset.id
    wx.navigateTo({ url: `/packages/show/pages/room/room?id=${id}` })
  },

  goPoints() {
    wx.switchTab({ url: '/pages/mine/mine' })
  }
})

Page({
  data: {
    products: [],
    points: 0,
    orders: []
  },

  unsub: null,

  onLoad() {
    this.unsub = store.subscribe((s) => this.setData({ points: s.points }))
    this.load()
  },

  onShow() {
    // 从其它页积分变动后回到本页即时刷新（订阅已覆盖，这里再拉一次订单）
    this.load()
  },

  onUnload() {
    if (this.unsub) this.unsub()
  },

  load() {
    api.getProducts().then((list) => this.setData({ products: list || [] }))
    api.getMallOrders().then((list) => this.setData({ orders: list || [] }))
  },

  onBuy(e) {
    const id = e.currentTarget.dataset.id
    const p = this.data.products.find((x) => x.id === id)
    if (!p) return
    wx.showModal({
      title: '确认兑换',
      content: '「' + p.title + '」需 ' + p.price + ' 积分，确认兑换？',
      confirmText: '兑换',
      confirmColor: '#C8161D',
      success: (r) => { if (r.confirm) this.doPurchase(p) }
    })
  },

  doPurchase(p) {
    api.purchaseProduct(p.id, 1).then((res) => {
      if (res && res.ok) {
        this.setData({ points: store.getState().points })
        api.getMallOrders().then((list) => this.setData({ orders: list || [] }))
        wx.showToast({ title: '兑换成功', icon: 'success' })
      } else if (res && res.reason === 'points') {
        wx.showToast({ title: '积分不足', icon: 'none' })
      } else {
        wx.showToast({ title: '兑换失败', icon: 'none' })
      }
    }).catch(() => wx.showToast({ title: '兑换失败', icon: 'none' }))
  }
})

// pages/read/read.js —— 我的阅读：已读篇章列表（本地阅读历史）
const res = require('../../services/resources.js')
const readHistory = require('../../services/readHistory.js')

const CAT_NAME = res.CATEGORIES.reduce((m, c) => { m[c.key] = c.name; return m }, {})
const READ_SHOW = 50 // 最多展示条数

const SWIPE_BTN = 72        // 删除按钮宽度（px）
const SWIPE_OPEN = -SWIPE_BTN   // 完全展开时的偏移
const SWIPE_THRESHOLD = 36  // 位移超过该阈值则自动展开（px）

Page({
  data: { list: [], count: 0 },

  // 拖拽过程临时状态（不放入 data，避免无谓渲染）
  _dragId: null,
  _startX: 0,
  _startOffset: 0,

  onShow() { this.refresh() },

  refresh() {
    const entries = readHistory.getEntries()
    const list = entries
      .map((e) => {
        const c = res.cards.find((x) => x.id === e.id)
        if (!c) return null
        return {
          id: c.id,
          title: c.title,
          subtitle: c.subtitle,
          catName: CAT_NAME[c.cat] || '',
          gradient: c.gradient,
          readTime: readHistory.formatDate(e.time),
          offset: 0,   // 右滑偏移（px，负数表示露出删除键）
          anim: false  // 偏移变化时是否启用过渡动画
        }
      })
      .filter(Boolean)
      .slice(0, READ_SHOW)
    this.setData({ list, count: entries.length })
  },

  idxOf(id) {
    return this.data.list.findIndex((x) => x.id === id)
  },

  onRead(e) {
    const id = e.currentTarget.dataset.id
    const i = this.idxOf(id)
    const item = this.data.list[i]
    // 已滑开（露出删除键）时，点击内容先收起而非跳转
    if (item && item.offset) {
      this.setData({ [`list[${i}].offset`]: 0, [`list[${i}].anim`]: true })
      return
    }
    wx.navigateTo({ url: '/pages/resource/detail/detail?id=' + id })
  },

  // —— 右滑删除手势 ——
  onSwipeStart(e) {
    const id = e.currentTarget.dataset.id
    const i = this.idxOf(id)
    if (i < 0) return
    this._dragId = id
    this._startX = e.touches[0].clientX
    this._startOffset = this.data.list[i].offset || 0
    // 拖拽中关闭过渡动画，并把其它已展开的项收起
    const updates = {}
    this.data.list.forEach((it, idx) => {
      if (it.id !== id && it.offset) updates[`list[${idx}].offset`] = 0
    })
    updates[`list[${i}].anim`] = false
    this.setData(updates)
  },

  onSwipeMove(e) {
    if (!this._dragId) return
    const i = this.idxOf(this._dragId)
    if (i < 0) return
    let offset = this._startOffset + (e.touches[0].clientX - this._startX)
    if (offset > 0) offset = 0
    if (offset < SWIPE_OPEN) offset = SWIPE_OPEN
    this.setData({ [`list[${i}].offset`]: offset })
  },

  onSwipeEnd(e) {
    if (!this._dragId) return
    const i = this.idxOf(this._dragId)
    this._dragId = null
    if (i < 0) return
    const offset = this.data.list[i].offset || 0
    const target = offset <= -SWIPE_THRESHOLD ? SWIPE_OPEN : 0
    this.setData({ [`list[${i}].offset`]: target, [`list[${i}].anim`]: true })
  },

  // 点击删除键：移除单条阅读记录
  onDelete(e) {
    const id = e.currentTarget.dataset.id
    readHistory.remove(id)
    this.refresh()
  },

  onClear() {
    wx.showModal({
      title: '清空阅读记录',
      content: '确定清空「我的阅读」记录？',
      success: (r) => { if (r.confirm) { readHistory.clear(); this.refresh() } }
    })
  }
})

  }

  },

  onRead(e) {
    const id = e.currentTarget.dataset.id

  }

const postcards = require('../../services/postcards.js')

Page({
  data: {
    list: [],

  },

  onShow() {
    loadFonts()
    this.load()
  },

  load() {
    const list = postcards.getList().map((p) => decorateCard(p))
    this.setData({ list })
  },

  onPreview(e) {
    const id = e.currentTarget.dataset.id
    const item = this.data.list.find((p) => p.id === id)
    if (item) this.setData({ preview: item })
  },

  onClosePreview() {
    this.setData({ preview: null })
  },

  // 列表长按删除
  onDeleteList(e) {
    const id = e.currentTarget.dataset.id
    const item = this.data.list.find((p) => p.id === id)
    if (!item) return
    wx.showModal({
      title: '删除作品',
      content: '确定删除这张明信片吗？删除后不可恢复。',
      confirmText: '删除',
      confirmColor: '#C8161D',
      success: (m) => {
        if (m.confirm) {
          postcards.remove(id)
          this.setData({ preview: null })
          this.load()
          wx.showToast({ title: '已删除', icon: 'success' })
        }
      }
    })
  },

  // 现绘明信片导出图片（无 imageUrl 时），返回 Promise<tempPath>
  // 导出统一为竖版(2:3)，横版内容顺时针旋转 90°（与展示态一致）
  drawRecord(rec) {
    return new Promise((resolve) => {
      if (!rec) return resolve('')
      wx.createSelectorQuery().in(this)
        .select('#pcCanvas')
        .fields({ node: true, size: true })
        .exec((res) => {
          if (!res || !res[0] || !res[0].node) return resolve('')
          const canvas = res[0].node
          renderPostcard(canvas, rec)
          wx.canvasToTempFilePath({
            canvas,
            success: (e) => resolve(e.tempFilePath),
            fail: () => resolve('')
          })
        })
    })
  },

  onSaveFromPreview() {
    const rec = this.data.preview
    if (!rec) return
    const save = (path) => {
      if (!path) { wx.showToast({ title: '稍候再试', icon: 'none' }); return }
      wx.saveImageToPhotosAlbum({
        filePath: path,
        success: () => wx.showToast({ title: '已保存相册', icon: 'success' }),
        fail: (err) => {
          if (/auth|deny/i.test(err.errMsg || '')) {
            wx.showModal({
              title: '需要相册权限',
              content: '请在设置中开启「保存到相册」权限',
              confirmText: '去设置',
              success: (m) => { if (m.confirm) wx.openSetting() }
            })
          } else {
            wx.showToast({ title: '保存失败', icon: 'none' })
          }
        }
      })
    }
    if (rec.imageUrl) { save(rec.imageUrl); return }
    this.drawRecord(rec).then((path) => save(path))
  },

  // 打印：绘制高清图后用系统图片查看器（可长按保存 / 打印）
  onPrint() {

    if (rec.imageUrl) { open(rec.imageUrl); return }
    this.drawRecord(rec).then(open)
  },

  onShareAppMessage() {
    const p = this.data.preview
    return {
      title: p ? ('我在红韵智演创作了「' + p.scene + '」明信片') : '红韵智演 · 我的红色明信片作品',
      path: '/pages/works/works',
      imageUrl: p && p.imageUrl ? p.imageUrl : ''
    }
  }
})

// 调用云函数的统一封装：成功取 result.data；result.code !== 0 视为业务错误
function call(name, data) {
  return new Promise((resolve, reject) => {
    wx.cloud.callFunction({
      name,
      data: data || {},
      success: (res) => {
        const r = res.result || {}
        if (r.code === 0) resolve(r.data)
        else reject(new Error(r.msg || ('cloud function ' + name + ' failed')))
      },
      fail: (err) => reject(err)
    })
  })
}

function delay(data, ms = 420) {
  return new Promise((resolve) => setTimeout(() => resolve(JSON.parse(JSON.stringify(data))), ms))
}

  return map.map((o) => ({ ...o, votes: o.votes + Math.floor(Math.random() * 6) }))
}

// services/favorites.js —— 本地阅读收藏（"阅读收藏"模块）
//
// 使用 wx 本地存储记录用户收藏的资料卡 id（倒序），无需后端依赖。
// 游客端未接入账号体系前，收藏仅保存在本机，卸载小程序或清缓存会丢失。

const KEY = 'rede_favorites'
const MAX = 200

function getList() {
  try {
    const v = wx.getStorageSync(KEY)
    return Array.isArray(v) ? v.slice() : []
  } catch (e) {
    return []
  }
}

function has(id) {
  return getList().indexOf(id) !== -1
}

// 切换收藏状态：已收藏则取消，未收藏则加入最前；返回切换后的收藏状态（true=已收藏）
function toggle(id) {
  if (!id) return false
  let list = getList()
  const i = list.indexOf(id)
  if (i !== -1) list.splice(i, 1)
  else {
    list.unshift(id)
    if (list.length > MAX) list = list.slice(0, MAX)
  }
  try { wx.setStorageSync(KEY, list) } catch (e) {}
  return list.indexOf(id) !== -1
}

function remove(id) {
  const list = getList().filter((x) => x !== id)
  try { wx.setStorageSync(KEY, list) } catch (e) {}
  return list
}

function clear() {
  try { wx.setStorageSync(KEY, []) } catch (e) {}
  return []
}

// 从 family 取首选字体族名（去掉引号），作为 loadFontFace 的 family
function firstFamily(family) {
  return family.split(',')[0].replace(/['"]/g, '').trim()
}

// 加载单个字体（f 需含可用 url 或 _url）
function loadOne(f) {
  const src = f.url || f._url
  if (!src) return
  wx.loadFontFace({
    family: firstFamily(f.family),
    source: 'url("' + src + '")',
    success() {},
    fail(err) {
      // 失败时保持系统字体回退，不阻塞渲染
      if (typeof console !== 'undefined') {
        console.warn('[fonts] 加载失败', firstFamily(f.family), err)
      }
    }
  })
}

// 拉起内置字体加载：静态 url 直接加载；cloudId 先换临时 https 地址（同一会话内缓存，避免重复云调用）
function loadFonts() {
  if (typeof wx === 'undefined' || !wx.loadFontFace) return

  const entries = Object.keys(FONTS).map((key) => ({ key, f: FONTS[key] }))
  const direct = entries.filter((e) => e.f.url)
  const cloud = entries.filter((e) => !e.f.url && e.f.cloudId)

  // 1) 静态 url 直接加载
  direct.forEach((e) => loadOne(e.f))

  // 2) 云存储 FileID → 临时 https 地址 → 加载
  if (cloud.length && wx.cloud && typeof wx.cloud.getTempFileURL === 'function') {
    const fileList = cloud.map((e) => e.f.cloudId)
    wx.cloud.getTempFileURL({
      fileList,
      success(res) {
        const urlMap = {}
        ;(res.fileList || []).forEach((item) => {
          if (item && item.fileID && item.tempFileURL) urlMap[item.fileID] = item.tempFileURL
        })
        cloud.forEach((e) => {
          const u = urlMap[e.f.cloudId]
          if (u) {
            e.f.url = u // 会话内缓存，下次 loadFonts 走 direct 分支，省一次云调用
            loadOne(e.f)
          }
        })
      },
      fail() {}
    })
  }
}

// services/pcLayout.js —— 明信片排版引擎（与页面解耦，三页共用）
//
// 职责：
// - buildQuote：根据 名言 / 方向 / 排列 / 位置 计算展示方案（换行断行 / 句尾去标点 / 续行缩进）
// - decorateCard：补全渲染字段（quoteLines / quoteFontSize / indent / sign / 统一格式 date）
// - paintPostcard：Canvas 精绘（9 宫格定位、字号自适应、落款双行、名句避让落款带）

const { FONTS } = require('./fonts.js')
const POS_FULL = { top: 1, bottom: 1, center: 1 } // 满宽位置（横排时可整行书写）
const DEFAULT_QUOTE = ""
const SIGN_BAND = 96   // 底部落款带预留高度（Canvas px）
const SCENE_BAND = 112 // 顶部场景名预留高度（Canvas px）

// 将文本做「均衡分行」：行数尽量少，且相邻行字数差不超过 1
function splitBalanced(text, perLine) {
  const n = Math.max(1, Math.ceil(text.length / perLine))
  const base = Math.floor(text.length / n)
  const rem = text.length - base * n
  const lines = []
  let i = 0
  for (let k = 0; k < n; k++) {
    const take = base + (k < rem ? 1 : 0)
    lines.push(text.substr(i, take))
    i += take
  }
  return lines
}

// 以换行符断行（换行本身不显示）；所有句子（行/列）结尾都不跟标点；过长子段再均衡拆分
function splitQuoteLines(text, arrange, perLine) {
  let segs = String(text).split(/\r?\n/).map(s => s.trim()).filter(s => s.length)
  if (!segs.length) segs = [String(text).trim() || String(text)]
  // 句尾不跟标点：去除段尾全角/半角标点（横竖排一致）
  segs = segs.map(s => s.replace(/[。！？!?；：、，,.…—~·]+$/, ''))
  const lines = []
  for (const s of segs) {
    if (s.length <= perLine) lines.push(s)
    else lines.push(...splitBalanced(s, perLine))
  }
  return lines
}

// 日期统一为 YYYY.MM.DD（落库为 YYYY-MM-DD，预览/导出一致）
function formatDate(d) {
  if (!d) return ''
  const m = String(d).match(/^(\d{4})-(\d{2})-(\d{2})$/)
  if (m) return m[1] + '.' + m[2] + '.' + m[3]
  return d
}

// 计算名言的展示方案（与 Canvas 规则对齐：换行断行 / 句尾去标点 / 续行缩进）
function buildQuote(quote, orient, arrange, position) {
  const raw = (quote && String(quote).trim()) || DEFAULT_QUOTE
  // 以换行符断行、句尾去标点；空行忽略
  let lines = String(raw).split(/\r?\n/).map(s => s.trim()).filter(s => s.length)
    .map(s => s.replace(/[。！？!?；：、，,.…—~·]+$/, ''))
  if (!lines.length) lines = [raw]
  const full = !!POS_FULL[position]
  const indent = (position === 'left' || position === 'tl' || position === 'bl')
  const longest = lines.reduce((m, s) => Math.max(m, s.length), 0)

  if (arrange === 'v') {
    // 竖排：以最长段定字号（尽量一列放下）
    const maxCol = orient === 'port' ? 16 : 11
    let fs = 34
    if (longest > maxCol) fs = Math.max(22, Math.round((34 * maxCol) / longest))
    return { lines, fontSizeRpx: fs, indent }
  }

  // 横排：按位置估算单行容量（基准字号 34rpx，CJK 约等宽）
  const perLine = orient === 'port'
    ? (full ? 10 : 5)
    : (full ? 15 : 7)
  if (longest <= perLine) {
    const boost = longest <= Math.ceil(perLine * 0.6) ? 6 : 0
    return { lines, fontSizeRpx: Math.min(44, 34 + boost), indent }
  }
  const fs = Math.max(22, Math.round((34 * perLine) / longest))
  return { lines, fontSizeRpx: fs, indent }
}

// 补全一张明信片渲染所需字段
function decorateCard(rec) {
  const orient = rec.orient || 'land'
  const arrange = rec.arrange || 'h'
  const position = rec.position || 'center'
  const q = buildQuote(rec.quote, orient, arrange, position)
  return Object.assign({}, rec, {
    orient, arrange, position,
    quoteLines: q.lines,
    quoteFontSize: q.fontSizeRpx,
    indent: q.indent,
    sign: rec.sign || '红韵智演',
    date: formatDate(rec.date)
  })
}

// ---------------- Canvas 精绘 ----------------

function paintPostcard(ctx, W, H, rec) {
  const colors = (rec.colors && rec.colors.length === 2) ? rec.colors : ['#C8161D', '#7A0A10']
  const F = FONTS[rec.font] || FONTS.maoti
  const orient = rec.orient || 'land'
  const arrange = rec.arrange || 'h'
  const position = rec.position || 'center'

  // 背景
  const g = ctx.createLinearGradient(0, 0, W, H)
  g.addColorStop(0, colors[0])
  g.addColorStop(1, colors[1])
  ctx.fillStyle = g
  ctx.fillRect(0, 0, W, H)

  ctx.fillStyle = '#ffffff'

  // 场景名（左上）
  ctx.textBaseline = 'top'
  ctx.textAlign = 'left'
  ctx.font = F.weight + ' 40px ' + F.family
  ctx.fillText(rec.scene || '', 40, 40)

  // 落款（右下，双行：日期 / 红韵智演）
  ctx.textAlign = 'right'
  ctx.fillStyle = 'rgba(255,255,255,.9)'
  ctx.font = '400 22px sans-serif'
  ctx.fillText(rec.date || '', W - 40, H - 76)
  ctx.fillStyle = '#ffffff'
  ctx.font = '700 28px sans-serif'
  ctx.fillText(rec.sign || '红韵智演', W - 40, H - 44)

  // 名言（不加引号，按用户要求纯文本呈现）
  const quote = (rec.quote && String(rec.quote).trim()) ? rec.quote : DEFAULT_QUOTE
  const q = quote
  if (arrange === 'v') paintVertical(ctx, q, W, H, F, orient, position)
  else paintHorizontal(ctx, q, W, H, F, orient, position)
}

function paintHorizontal(ctx, q, W, H, F, orient, position) {
  const pad = 40
  // br（右下）多行时降级为 bottom 居中，避让右下落款带
  const eff = (position === 'br' && q.length > (orient === 'port' ? 9 : 14)) ? 'bottom' : position

  let boxX = pad
  let boxW = W - pad * 2
  let align = 'center'
  if (eff === 'left' || eff === 'tl' || eff === 'bl') { boxW = (W - pad * 2) * 0.62; align = 'left' }
  if (eff === 'right' || eff === 'tr') { boxX = pad + (W - pad * 2) * 0.38; boxW = (W - pad * 2) * 0.62; align = 'right' }

  // 字号自适应：先按换行切段、去句尾标点，以最长段为基准测量，超宽则缩（min 20）
  let fs = orient === 'port' ? 40 : 38
  ctx.font = F.weight + ' ' + fs + 'px ' + F.family
  const segs = splitQuoteLines(q, 'h', q.length)
  let longestW = 0
  for (const s of segs) longestW = Math.max(longestW, ctx.measureText(s).width)
  if (longestW > boxW) {
    fs = Math.max(20, Math.floor((fs * boxW) / longestW))
    ctx.font = F.weight + ' ' + fs + 'px ' + F.family
  }

  // 分行：以换行符断行；续行缩进两字；超长段再均衡折行
  const perLine = Math.max(1, Math.floor(boxW / (fs * 0.95)))
  let lines = splitQuoteLines(q, 'h', perLine)

  // 限行（最多 6 行）：超出再缩字号
  if (lines.length > 6) {
    fs = Math.max(18, Math.floor((fs * 6) / lines.length))
    ctx.font = F.weight + ' ' + fs + 'px ' + F.family
    const pl = Math.max(1, Math.floor(boxW / (fs * 0.95)))
    lines = splitQuoteLines(q, 'h', pl)
    if (lines.length > 6) lines = lines.slice(0, 6)
  }

  const lineH = fs * 1.5
  const totalH = lines.length * lineH
  const indentX = fs * 2 // 续行缩进两字

  // 垂直锚点（bottom 系预留落款带）
  let y
  if (eff === 'top' || eff === 'tl' || eff === 'tr') y = SCENE_BAND
  else if (eff === 'bottom' || eff === 'bl' || eff === 'br') y = H - SIGN_BAND - totalH
  else y = (H - totalH) / 2

  ctx.textBaseline = 'top'
  ctx.textAlign = align
  ctx.fillStyle = '#ffffff'
  for (let i = 0; i < lines.length; i++) {
    const baseX = align === 'center' ? boxX + boxW / 2
      : align === 'right' ? boxX + boxW
      : boxX
    // 续行相对上一行缩进两字（右对齐则向左缩）
    const x = baseX + (i > 0 ? (align === 'right' ? -indentX : indentX) : 0)
    ctx.fillText(lines[i], x, y + i * lineH)
  }
}

function paintVertical(ctx, q, W, H, F, orient, position) {
  const pad = 40
  // 横版竖向容量放宽，使单列可容纳约 14 汉字 + 1 逗号（约 15 字）；竖版保持更大留白
  const maxH = orient === 'port' ? H - 200 : H - 240
  // 先按换行切段、去段尾标点，以最长段为基准定字号（保证单列不超列高）
  const segs = splitQuoteLines(q, 'v', q.length)
  const maxSeg = segs.reduce((m, s) => Math.max(m, s.length), 0)
  let fs = 46
  const colCap = Math.floor(maxH / (fs * 1.05))
  if (maxSeg > colCap) fs = Math.max(22, Math.floor((fs * colCap) / maxSeg))
  ctx.font = F.weight + ' ' + fs + 'px ' + F.family
  const charH = fs * 1.05
  const cap = Math.floor(maxH / charH)

  // 每段再按列高切列（列序：续列朝中心方向展开）
  const cols = []
  for (const seg of segs) {
    let i = 0
    while (i < seg.length) { cols.push(seg.substr(i, cap)); i += cap }
  }

  // 水平锚点
  let x, colDir
  if (position === 'left' || position === 'tl' || position === 'bl') {
    x = pad + fs / 2
    colDir = 1
  } else if (position === 'center') {
    // 居中：以首列（最右列）中心为基准，整块列向中心展开后整体水平居中
    x = W / 2 + (cols.length - 1) * (fs * 1.15) / 2
    colDir = -1
  } else {
    x = W - pad - fs / 2
    colDir = -1
  }

  // 续列（换行后 / 段内换行）整体向下缩进两字
  const indentY = fs * 2
  const mainH = (cols[0] ? cols[0].length * charH : 0) + indentY

  // 垂直锚点（bottom 系预留落款带；以首列高度 + 续列缩进估算整体块高）
  let y0
  if (position === 'top' || position === 'tl' || position === 'tr') y0 = SCENE_BAND
  else if (position === 'bottom' || position === 'bl' || position === 'br') y0 = H - SIGN_BAND - mainH
  else y0 = (H - mainH) / 2

  // 竖版竖排（左锚点）单列下移，避让左上角场景名，不与其相接
  if (orient === 'port' && (position === 'left' || position === 'tl' || position === 'bl')) {
    y0 = Math.max(y0, SCENE_BAND + 16)
  }

  ctx.textBaseline = 'top'
  ctx.textAlign = 'center'
  ctx.fillStyle = '#ffffff'
  for (let c = 0; c < cols.length; c++) {
    const cx = x + colDir * c * (fs * 1.15)
    const col = cols[c]
    const yStart = y0 + (c > 0 ? indentY : 0)
    for (let j = 0; j < col.length; j++) {
      ctx.fillText(col[j], cx, yStart + j * charH)
    }
  }
}

// 把明信片绘制到离屏 canvas 并导出：统一为竖版(2:3)，
// 横版内容顺时针旋转 90° CW（展示态与导出均统一竖版 2:3、横版内旋），所见即所得。
// canvas：type=2d 的 canvas node；card：与 paintPostcard 同结构。
function renderPostcard(canvas, card) {
  const orient = (card && card.orient) || 'land'
  const dpr = (wx.getWindowInfo ? wx.getWindowInfo().pixelRatio : wx.getSystemInfoSync().pixelRatio) || 2
  const W = 600
  const H = 900
  canvas.width = W * dpr
  canvas.height = H * dpr
  const ctx = canvas.getContext('2d')
  // 重设 canvas.width 已重置变换矩阵；这里再显式归零，复用同一节点时安全
  ctx.setTransform(1, 0, 0, 1, 0, 0)
  ctx.clearRect(0, 0, canvas.width, canvas.height)
  ctx.scale(dpr, dpr)
  const c = decorateCard(card)
  if (orient === 'land') {
    // 横版 3:2 组合旋转 90° CW 放入 600×900 竖版画布：
    // 逻辑 900×600 → translate(600,0) + rotate(π/2) 恰好映射到 600×900
    ctx.translate(W, 0)
    ctx.rotate(Math.PI / 2)
    paintPostcard(ctx, 900, 600, c)
  } else {
    paintPostcard(ctx, W, H, c)
  }
  return canvas
}

module.exports = { buildQuote, decorateCard, paintPostcard, renderPostcard, DEFAULT_QUOTE }

// services/postcards.js —— 本地「我的作品」明信片存储
// 与 readHistory / favorites 一致：用 wx 本地存储持久化用户创作的明信片，
// 保证「已存入我的作品」提示真实可信，且创作页/我的页/作品列表页同源读取。

const KEY = 'rede_postcards'

function getList() {
  try {
    const v = wx.getStorageSync(KEY)
    return Array.isArray(v) ? v : []
  } catch (e) {
    return []
  }
}

function today() {
  const d = new Date()
  const p = (n) => (n < 10 ? '0' + n : '' + n)
  return d.getFullYear() + '-' + p(d.getMonth() + 1) + '-' + p(d.getDate())
}

// 新增一条作品（置顶），自动补 id / date
function add(rec) {
  const list = getList()
  const item = Object.assign({ id: 'p' + Date.now(), date: today() }, rec)
  list.unshift(item)
  try { wx.setStorageSync(KEY, list) } catch (e) {}
  return list
}

function remove(id) {
  const list = getList().filter((p) => p.id !== id)
  try { wx.setStorageSync(KEY, list) } catch (e) {}
  return list
}

function clear() {
  try { wx.setStorageSync(KEY, []) } catch (e) {}
  return []
}

// 存储键是否曾被写入：用于区分"从未初始化"与"用户已主动删空"
//（WeChat 中未设置的键 getStorageSync 返回 ''，已设为空数组则返回 []）
function isInitialized() {
  try {
    return wx.getStorageSync(KEY) !== ''
  } catch (e) {
    return false
  }
}

module.exports = { getList, add, remove, clear, today, isInitialized }

/* ============ services/readHistory.js ============ */
// services/readHistory.js —— 本地阅读历史（"我的阅读"模块）
//
// 使用 wx 本地存储记录已读资料卡（含阅读时间），倒序、去重、限量，无需后端依赖。
// 游客端未接入账号体系前，阅读记录仅保存在本机，卸载小程序或清缓存会丢失。

const KEY = 'rede_read_history'
const MAX = 50

function getRaw() {
  try {
    const v = wx.getStorageSync(KEY)
    return Array.isArray(v) ? v : []
  } catch (e) {
    return []
  }
}

// 兼容旧数据（旧版仅存 id 字符串）：统一为 { id, time }
function normalize(list) {
  return list.map((e) => {
    if (typeof e === 'string') return { id: e, time: Date.now() }
    return { id: e.id, time: e.time || Date.now() }
  })
}

// 已读 id 列表（最新在前），供去重 / 已读状态判断使用
function getList() {
  return normalize(getRaw()).map((e) => e.id)
}

// 已读条目（含阅读时间），供列表展示使用
function getEntries() {
  return normalize(getRaw())
}

// 记录一次阅读：把 id 提到最前、刷新阅读时间、去重、限量
function add(id) {
  if (!id) return getEntries()
  const list = getRaw().filter((e) => (typeof e === 'string' ? e : e.id) !== id)
  list.unshift({ id, time: Date.now() })
  if (list.length > MAX) list = list.slice(0, MAX)
  try { wx.setStorageSync(KEY, list) } catch (e) {}
  return getEntries()
}

function clear() {
  try { wx.setStorageSync(KEY, []) } catch (e) {}
  return []
}

// 阅读时间格式化：2026 年 7 月 19 日  14 : 30（24 小时制）
function pad(n) {
  return n < 10 ? '0' + n : '' + n
}

function formatDate(time) {
  const d = new Date(time)
  const y = d.getFullYear()
  const m = d.getMonth() + 1
  const day = pad(d.getDate())
  const hh = pad(d.getHours())
  const mm = pad(d.getMinutes())
  return y + ' 年 ' + m + ' 月 ' + day + ' 日  ' + hh + ' : ' + mm
}

// 删除单条阅读记录（按 id），返回删除后的条目列表
function remove(id) {
  const list = getRaw().filter((e) => (typeof e === 'string' ? e : e.id) !== id)
  try { wx.setStorageSync(KEY, list) } catch (e) {}
  return getEntries()
}

module.exports = { getList, getEntries, add, remove, clear, formatDate }

// services/rewards.js —— 积分奖励与明信片计费
//
// 规则（与产品需求一致）：
//   1) 阅读一篇文章   +1 积分（同一篇只加一次，按已奖励集合去重）
//   2) 参加一次演出   +5 积分（同一场只加一次）
//   3) 明信片前 5 次生成免费；之后每次生成消耗 1 积分
//      - 免费额度用「累计生成次数」判定，删除作品不会回退已用掉的免费次数
//      - 计费用独立的生成计数器，避免「删了再生成」刷免费额度

const store = require('../store.js')
const storage = require('./storage.js')

const READ_KEY = 'rede_awarded_reads'   // 已奖励积分的文章 id 集合
const SHOW_KEY = 'rede_awarded_shows'   // 已奖励积分的演出 id 集合
const GEN_KEY = 'rede_postcard_count'   // 累计明信片生成次数
const POSTCARD_FREE = 5                 // 前 N 次免费

function getArr(key) {
  const v = storage.get(key, [])
  return Array.isArray(v) ? v : []
}
function pushArr(key, id) {
  const a = getArr(key)
  if (!a.includes(id)) {
    a.push(id)
    storage.set(key, a)
  }
}

// 阅读一篇文章 +1（同篇一次）
function awardRead(id) {
  if (!id) return false
  if (getArr(READ_KEY).includes(id)) return false
  pushArr(READ_KEY, id)
  store.addPoints(1)
  return true
}

// 参加一次演出 +5（同场一次）
function awardShow(showId) {
  if (!showId) return false
  if (getArr(SHOW_KEY).includes(showId)) return false
  pushArr(SHOW_KEY, showId)
  store.addPoints(5)
  return true
}

// 当前明信片计费状态（生成前查询，用于 UI 提示与拦截）
// 返回 { count, free, freeLeft }
function postcardState() {
  const n = getGenCount()
  return {
    count: n,
    free: n < POSTCARD_FREE,
    freeLeft: Math.max(0, POSTCARD_FREE - n)
  }
}

// 生成成功后提交：计数 +1；若已超过免费次数则扣 1 积分
// 返回 { free } —— 本次是否免费
function commitPostcard() {
  const n = getGenCount()
  const free = n < POSTCARD_FREE
  storage.set(GEN_KEY, n + 1)
  if (!free) store.addPoints(-1)
  return { free }
}

function getGenCount() {
  const v = storage.get(GEN_KEY, 0)
  return typeof v === 'number' ? v : 0
}

module.exports = { awardRead, awardShow, postcardState, commitPostcard, POSTCARD_FREE }

// services/storage.js —— 安全的 wx 本地存储封装
// 统一处理「键未设置」「非小程序环境（node 桩测）」等边界，避免各处重复写 try/catch。

function get(key, fallback) {
  try {
    if (typeof wx !== 'undefined' && wx.getStorageSync) {
      const v = wx.getStorageSync(key)
      if (v === '' || v === undefined || v === null) return fallback
      return v
    }
  } catch (e) {}
  return fallback
}

function set(key, val) {
  try {
    if (typeof wx !== 'undefined' && wx.setStorageSync) {
      wx.setStorageSync(key, val)
    }
  } catch (e) {}
}

module.exports = { get, set }

/* ============ store.js ============ */
// store.js —— 轻量全局状态（用户 / 积分 / 今日获得 / 身份）
// 页面通过 subscribe 订阅，onUnload 时取消订阅，避免内存泄漏。
// 真实环境由云函数 login 写入 userInfo，这里先用本地兜底。
//
// 积分持久化：points 与 todayGain（今日获得）写入本地存储，跨页面 / 重启保留；
// todayGain 按自然日自动清零，仅统计当日通过奖励获得的积分（消费不计入）。

const storage = require('./services/storage.js')

const POINTS_KEY = 'rede_points'
const TODAY_KEY = 'rede_today'
const SEED_POINTS = 1280

function todayStr() {
  const d = new Date()
  const p = (n) => (n < 10 ? '0' + n : '' + n)
  return d.getFullYear() + '-' + p(d.getMonth() + 1) + '-' + p(d.getDate())
}

function loadPoints() {
  const v = storage.get(POINTS_KEY, null)
  if (typeof v === 'number') return v
  storage.set(POINTS_KEY, SEED_POINTS) // 首次使用：种子积分
  return SEED_POINTS
}

function loadToday() {
  const v = storage.get(TODAY_KEY, null)
  if (v && typeof v === 'object' && v.date === todayStr()) {
    return typeof v.gain === 'number' ? v.gain : 0
  }
  return 0
}

const state = {
  user: null,        // { openid, nickname, avatar, level, isStaff }
  points: loadPoints(),         // 当前积分余额（持久化）
  isStaff: false,    // 是否为工作人员（场务/演员）
  loaded: false,
  todayGain: loadToday(),       // 今日获得的积分（持久化，按日清零）
  todayDate: todayStr()
}

const subs = new Set()

function notify() {
  subs.forEach((fn) => {
    try { fn(state) } catch (e) { console.error('[store] listener error', e) }
  })
}

// 跨日则重置今日获得
function ensureToday() {
  const t = todayStr()
  if (state.todayDate !== t) {
    state.todayDate = t
    state.todayGain = 0
    storage.set(TODAY_KEY, { date: t, gain: 0 })
  }
}

module.exports = {
  getState() { return state },

  setUser(user) {
    Object.assign(state, user)
    state.loaded = true
    notify()
  },

  addPoints(delta) {
    state.points = Math.max(0, state.points + delta)
    // 仅正向奖励累计「今日获得」；消费（负向）不影响今日获得
    if (delta > 0) {
      ensureToday()
      state.todayGain += delta
      storage.set(TODAY_KEY, { date: state.todayDate, gain: state.todayGain })
    }
    storage.set(POINTS_KEY, state.points)
    notify()
  },

  setPoints(points) {
    state.points = points
    storage.set(POINTS_KEY, state.points)
    notify()
  },

  subscribe(fn) {
    subs.add(fn)
    // 立即回放一次当前状态
    fn(state)
    return () => subs.delete(fn)
  },

  reset() {
    state.user = null
    state.points = 0
    state.isStaff = false
    state.loaded = false
    storage.set(POINTS_KEY, 0)
    storage.set(TODAY_KEY, { date: todayStr(), gain: 0 })
    notify()
  }
}
